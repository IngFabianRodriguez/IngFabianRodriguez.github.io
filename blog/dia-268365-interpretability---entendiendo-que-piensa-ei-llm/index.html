🎯 La Matemática del Costo

Antes de optimizar, hay que entender cómo se calcula el costo:

Costo Total = (Tokens Input × Costo por Token Input) + (Tokens Output × Costo por Token Output)

Ejemplo con GPT-4o:

  • Input: $5.00 / 1M tokens
  • Output: $15.00 / 1M tokens
  • Un chat típico de 1000 tokens input + 500 tokens output = $0.005 + $0.0075 = $0.0125 por request

💡 Si tienes 100K requests diarios, eso es $1,250/día = $37,500/mes. Una optimización del 30% te ahorra $11,250/mes.

Para modelos auto-alojados:

Costo por Token = (Costo GPU-hora / Tokens procesados por hora)

Una A100 a $2/hora procesa ~10K tokens/segundo en batch:

= 10,000 × 3600 = 36M tokens/hora

= $2 / 36M = $0.000055 por 1K tokens

📊 Cost Tracking: Midiendo el Gasto

El error #1 es no saber cuánto gastas por request. Vamos a implementar tracking:

from dataclasses import dataclass, field
from datetime import datetime
from typing import Optional, Dict
import hashlib
import json

@dataclass
class CostRecord:
    """Registro de costo para una request."""
    request_id: str
    timestamp: str
    model: str
    prompt_tokens: int
    completion_tokens: int
    cost_usd: float
    cache_hit: bool = False
    metadata: Dict = field(default_factory=dict)

class CostTracker:
    """Tracker de costos para LLM inference."""

    def __init__(self, model_costs: Dict[str, Dict[str, float]]):
        """
        model_costs: {
            "gpt-4o": {"input_per_1k": 0.005, "output_per_1k": 0.015},
            "gpt-3.5-turbo": {"input_per_1k": 0.0005, "output_per_1k": 0.0015},
            "llama-2-70b": {"per_1k_tokens": 0.00007},  # costo por 1K tokens (ambos)
        }
        """
        self.model_costs = model_costs
        self.records: list[CostRecord] = []

    def calculate_cost(
        self,
        model: str,
        prompt_tokens: int,
        completion_tokens: int,
        cache_hit: bool = False
    ) -> float:
        """Calcula el costo en USD."""
        if cache_hit:
            # Cache hit tiene descuento (OpenAI cobra 10% para cached tokens)
            return 0.0

        costs = self.model_costs.get(model, {})
        if "input_per_1k" in costs:
            # Modelo con costos separados de input/output
            input_cost = (prompt_tokens / 1000) * costs["input_per_1k"]
            output_cost = (completion_tokens / 1000) * costs["output_per_1k"]
            return input_cost + output_cost
        elif "per_1k_tokens" in costs:
            # Costo uniforme por token
            total_tokens = prompt_tokens + completion_tokens
            return (total_tokens / 1000) * costs["per_1k_tokens"]
        else:
            return 0.0

    def log_request(
        self,
        model: str,
        prompt_tokens: int,
        completion_tokens: int,
        cache_hit: bool = False,
        metadata: Optional[Dict] = None
    ) -> CostRecord:
        """Registra una request y su costo."""
        cost = self.calculate_cost(model, prompt_tokens, completion_tokens, cache_hit)

        record = CostRecord(
            request_id=hashlib.md5(f"{datetime.now()}{prompt_tokens}".encode()).hexdigest()[:12],
            timestamp=datetime.now().isoformat(),
            model=model,
            prompt_tokens=prompt_tokens,
            completion_tokens=completion_tokens,
            cost_usd=cost,
            cache_hit=cache_hit,
            metadata=metadata or {}
        )

        self.records.append(record)
        return record

    def summary(self) -> Dict:
        """Resumen de costos."""
        if not self.records:
            return {"total_cost": 0, "total_requests": 0}

        total = sum(r.cost_usd for r in self.records)
        cache_hits = sum(1 for r in self.records if r.cache_hit)
        by_model = {}
        for r in self.records:
            by_model[r.model] = by_model.get(r.model, 0) + r.cost_usd

        return {
            "total_cost_usd": round(total, 4),
            "total_requests": len(self.records),
            "cache_hit_rate": round(cache_hits / len(self.records) * 100, 1),
            "cost_by_model": {k: round(v, 4) for k, v in by_model.items()},
            "avg_cost_per_request": round(total / len(self.records), 6)
        }

    def export_csv(self, filepath: str):
        """Exporta a CSV para análisis."""
        import csv
        with open(filepath, "w", newline="") as f:
            writer = csv.DictWriter(f, fieldnames=[
                "request_id", "timestamp", "model", "prompt_tokens",
                "completion_tokens", "cost_usd", "cache_hit"
            ])
            writer.writeheader()
            for r in self.records:
                writer.writerow({
                    "request_id": r.request_id,
                    "timestamp": r.timestamp,
                    "model": r.model,
                    "prompt_tokens": r.prompt_tokens,
                    "completion_tokens": r.completion_tokens,
                    "cost_usd": r.cost_usd,
                    "cache_hit": r.cache_hit
                })

# Uso
tracker = CostTracker({
    "gpt-4o": {"input_per_1k": 0.005, "output_per_1k": 0.015},
    "gpt-3.5-turbo": {"input_per_1k": 0.0005, "output_per_1k": 0.0015},
    "llama-2-70b": {"per_1k_tokens": 0.00007},
})

# Registrar request
record = tracker.log_request(
    model="gpt-4o",
    prompt_tokens=500,
    completion_tokens=200,
    metadata={"user_id": "user_123", "endpoint": "/chat"}
)

print(f"Costo: ${record.cost_usd:.4f}")
print(tracker.summary())

🔀 Model Routing: Ahorra 60% con Inteligencia

No toda request necesita GPT-4. Clasifica y enruta:

from enum import Enum
from typing import Literal

class TaskComplexity(Enum):
    SIMPLE = "simple"      # Preguntas cortas, respuestas cortas
    MEDIUM = "medium"      # Análisis moderado, código estándar
    COMPLEX = "complex"    # Razonamiento profundo, contexto largo

class ModelRouter:
    """Router inteligente de requests a modelos apropiados."""

    def __init__(self, tracker: CostTracker):
        self.tracker = tracker
        # Modelo pequeño: rápido y barato
        self.small_model = "gpt-3.5-turbo"
        # Modelo grande: caro pero poderoso
        self.large_model = "gpt-4o"
        # Umbrales
        self.simple_max_tokens = 50
        self.medium_max_tokens = 300

    def classify_task(
        self,
        prompt: str,
        max_response_tokens: int = 100
    ) -> TaskComplexity:
        """Clasifica la complejidad de la tarea."""
        # Señales de tarea simple
        simple_indicators = [
            len(prompt) < 50,
            max_response_tokens < self.simple_max_tokens,
            any(kw in prompt.lower() for kw in ["que es", "define", "dime la hora"]),
        ]

        # Señales de tarea compleja
        complex_indicators = [
            len(prompt) > 500,
            max_response_tokens > self.medium_max_tokens,
            any(kw in prompt.lower() for kw in ["analiza", "compara", "explica en profundidad"]),
            "código" in prompt.lower() and "complejo" in prompt.lower(),
        ]

        if sum(simple_indicators) >= 2 and not any(complex_indicators):
            return TaskComplexity.SIMPLE
        elif any(complex_indicators):
            return TaskComplexity.COMPLEX
        else:
            return TaskComplexity.MEDIUM

    def route(
        self,
        prompt: str,
        max_response_tokens: int = 100,
        force_model: Optional[str] = None
    ) -> Literal["gpt-3.5-turbo", "gpt-4o"]:
        """Determina qué modelo usar."""
        if force_model:
            return force_model

        complexity = self.classify_task(prompt, max_response_tokens)

        if complexity == TaskComplexity.SIMPLE:
            return self.small_model
        elif complexity == TaskComplexity.COMPLEX:
            return self.large_model
        else:
            # Medium: depende del presupuesto
            return self.small_model  # Por defecto el barato

    def execute_with_routing(
        self,
        prompt: str,
        max_response_tokens: int = 100,
        client=None  # OpenAI client
    ) -> Dict:
        """Ejecuta request con routing inteligente."""
        model = self.route(prompt, max_response_tokens)

        # Simular llamada
        # response = client.chat.completions.create(
        #     model=model,
        #     messages=[{"role": "user", "content": prompt}],
        #     max_tokens=max_response_tokens
        # )

        # Registrar costo
        record = self.tracker.log_request(
            model=model,
            prompt_tokens=len(prompt) // 4,  # Estimación
            completion_tokens=max_response_tokens // 4,
            metadata={"complexity": self.classify_task(prompt, max_response_tokens).value}
        )

        return {
            "model_used": model,
            "cost": record.cost_usd,
            "routing_savings": record.cost_usd * 0.6 if model == self.small_model else 0
        }

# Usage
router = ModelRouter(tracker)

# Tareas simples van a modelo barato
result = router.execute_with_routing(
    prompt="Que es Python?",
    max_response_tokens=50
)
print(f"Modelo: {result['model_used']}, Costo: ${result['cost']:.6f}")  # gpt-3.5-turbo

# Tareas complejas van a modelo grande
result = router.execute_with_routing(
    prompt="Analiza el impacto de la IA en la economía global, considerando factores históricos, tecnológicos y sociales, con datos específicos.",
    max_response_tokens=1000
)
print(f"Modelo: {result['model_used']}, Costo: ${result['cost']:.6f}")  # gpt-4o

# Savings acumulado
print(f"Ahorro por routing: ${sum(r['routing_savings'] for r in [result]):.2f}")

💾 Semantic Caching: 30% de Requests Son Duplicados

El 30% de las requests son idénticas o muy similares. Cachealas:

import hashlib
from typing import Optional, Dict, List
import numpy as np

class SemanticCache:
    """Cache semántico para requests LLM."""

    def __init__(self, similarity_threshold: float = 0.95, embedding_model=None):
        """
        similarity_threshold: 0.0-1.0, qué tan similar debe ser para considerarse hit
        embedding_model: modelo para generar embeddings (e.g., sentence-transformers)
        """
        self.similarity_threshold = similarity_threshold
        self.embedding_model = embedding_model
        self.cache: Dict[str, Dict] = {}
        self.embedding_model = embedding_model

    def _get_embedding(self, text: str) -> np.ndarray:
        """Genera embedding para el texto."""
        if self.embedding_model:
            return self.embedding_model.encode(text)
        else:
            # Fallback: hash simple (no semántico, pero funciona)
            return np.array([hash(text) % 1000])

    def _cosine_similarity(self, a: np.ndarray, b: np.ndarray) -> float:
        """Calcula similitud coseno entre dos vectores."""
        return float(np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b)))

    def _compute_hash(self, text: str) -> str:
        """Compute a hash key for exact match."""
        return hashlib.sha256(text.encode()).hexdigest()[:16]

    def get(self, prompt: str, max_response_tokens: Optional[int] = None) -> Optional[str]:
        """Busca en cache. Retorna respuesta si hay hit, None si miss."""
        exact_key = self._compute_hash(prompt)

        # 1. Buscar por hash exacto primero
        if exact_key in self.cache:
            entry = self.cache[exact_key]
            if max_response_tokens is None or entry.get("max_tokens") == max_response_tokens:
                entry["hits"] += 1
                entry["last_accessed"] = "now"
                return entry["response"]

        # 2. Buscar por similitud semántica
        if self.embedding_model:
            prompt_embedding = self._get_embedding(prompt)

            best_match = None
            best_similarity = 0

            for key, entry in self.cache.items():
                if "embedding" in entry:
                    similarity = self._cosine_similarity(prompt_embedding, entry["embedding"])
                    if similarity > best_similarity and similarity >= self.similarity_threshold:
                        best_similarity = similarity
                        best_match = (key, entry)

            if best_match:
                key, entry = best_match
                entry["hits"] += 1
                entry["last_accessed"] = "now"
                return entry["response"]

        return None

    def put(
        self,
        prompt: str,
        response: str,
        max_response_tokens: Optional[int] = None,
        metadata: Optional[Dict] = None
    ):
        """Guarda en cache."""
        key = self._compute_hash(prompt)

        embedding = None
        if self.embedding_model:
            embedding = self._get_embedding(prompt)

        self.cache[key] = {
            "prompt": prompt,
            "response": response,
            "max_tokens": max_response_tokens,
            "hits": 0,
            "created_at": "now",
            "last_accessed": "now",
            "embedding": embedding,
            "metadata": metadata or {}
        }

    def stats(self) -> Dict:
        """Estadísticas del cache."""
        total_hits = sum(e["hits"] for e in self.cache.values())
        total_requests = total_hits + len(self.cache)  # hits + misses (estimado)

        return {
            "cache_size": len(self.cache),
            "total_hits": total_hits,
            "hit_rate": round(total_hits / total_requests * 100, 1) if total_requests > 0 else 0,
            "estimated_savings_usd": round(total_hits * 0.005, 2)  # Asumiendo $0.005 por request
        }

# Uso
from sentence_transformers import SentenceTransformer

embedder = SentenceTransformer('all-MiniLM-L6-v2')
cache = SemanticCache(similarity_threshold=0.92, embedding_model=embedder)

# Primera vez: cache miss
prompt = "Explica qué es machine learning en términos simples"
cached_response = cache.get(prompt)

if not cached_response:
    # response = call_llm(prompt)
    response = "Machine learning es una rama de la IA que permite a las máquinas aprender de datos..."
    cache.put(prompt, response, max_response_tokens=100)

    print("Cache MISS - guardando respuesta")
else:
    print(f"Cache HIT - respuesta: {cached_response}")

# Ver estadísticas
print(cache.stats())
# {'cache_size': 1, 'total_hits': 0, 'hit_rate': 0.0, 'estimated_savings_usd': 0.0}

⚖️ Cuantización: Mismo Modelo, Menor Costo

Reduce el costo de auto-alojamiento con cuantización:

# De 70B float16 (140GB) a 8-bit (70GB) - media GPU en vez de 2
# De 70B float16 a 4-bit (35GB) - una sola GPU

from transformers import AutoModelForCausalLM, BitsAndBytesConfig
import torch

# Cuantización 8-bit
model_8bit = AutoModelForCausalLM.from_pretrained(
    "meta-llama/Llama-2-70b-hf",
    quantization_config=BitsAndBytesConfig(
        load_in_8bit=True,
        llm_int8_threshold=6.0,  # Umbral para outlier activation
        llm_int8_skip_modules=["lm_head"],
    ),
    device_map="auto"
)

# Cuantización 4-bit (más agresiva, más ahorro)
model_4bit = AutoModelForCausalLM.from_pretrained(
    "meta-llama/Llama-2-70b-hf",
    quantization_config=BitsAndBytesConfig(
        load_in_4bit=True,
        bnb_4bit_compute_dtype=torch.float16,
        bnb_4bit_use_double_quant=True,
        bnb_4bit_quant_type="nf4"  # Normal Float 4-bit
    ),
    device_map="auto"
)

# El resultado: puedes correr 70B en UNA A100 de 80GB en vez de 2
# Costo: de ~$4/hora (2 GPUs) a ~$2/hora (1 GPU) = 50% ahorro

⚠️ Los 3 Errores Clásicos (y sus fixes)

Error 1: No trackear costo por request

Síntoma: "Somos $50K/mes" sin saber en qué. No hay visibilidad de qué endpoints, usuarios o features son los más costosos.

Fix: Implementa CostTracker desde día 1. Etiqueta cada request con metadata útil.

# Muestra por endpoint
costs_by_endpoint = {}
for record in tracker.records:
    endpoint = record.metadata.get("endpoint", "unknown")
    costs_by_endpoint[endpoint] = costs_by_endpoint.get(endpoint, 0) + record.cost_usd

for endpoint, cost in sorted(costs_by_endpoint.items(), key=lambda x: -x[1]):
    print(f"{endpoint}: ${cost:.2f}")

Error 2: Usar el mismo modelo para todo

Síntoma: GPT-4 para "¿Qué hora es?" y para "Análisis financiero completo". Estás quemando dinero.

Fix: Implementa model routing. 60% de requests pueden ir a modelo más barato.

💡 En producción real, típicamente: 50-60% tareas simples (gpt-3.5), 30-40% medium (gpt-4), 5-10% complex (gpt-4 con contexto largo). Routing automático puede reducir tu factura 40-60%.

Error 3: No usar caching

Síntoma: El mismo "¿Cómo reseteo mi contraseña?" 500 veces al día. Cada una cuesta $0.01. Eso es $5/día = $150/mes en requests idénticas.

Fix: Implementa semantic caching con embeddings. 30%+ hit rate es normal.

# Antes: sin cache
for user_query in user_queries:
    response = call_llm(user_query)  # $0.01 cada vez
    total_cost += 0.01

# Después: con cache
cache = SemanticCache(embedding_model=embedder)
for user_query in user_queries:
    cached = cache.get(user_query)
    if cached:
        response = cached
        savings += 0.01  # Lo que te ahorraste
    else:
        response = call_llm(user_query)
        cache.put(user_query, response)

print(f"Cache hit rate: {cache.stats()['hit_rate']}%")
print(f"Ahorro total: ${savings:.2f}")

📈 Cost Dashboard

# Dashboard para monitorear costos en producción
from datetime import datetime, timedelta

class CostDashboard:
    def __init__(self, tracker: CostTracker):
        self.tracker = tracker

    def daily_summary(self, days: int = 7) -> Dict:
        """Resumen de costos por día."""
        # Agrupar por día
        daily = {}
        for record in self.tracker.records:
            day = record.timestamp[:10]  # YYYY-MM-DD
            if day not in daily:
                daily[day] = {"cost": 0, "requests": 0, "cache_hits": 0}
            daily[day]["cost"] += record.cost_usd
            daily[day]["requests"] += 1
            if record.cache_hit:
                daily[day]["cache_hits"] += 1

        return {
            day: {
                "cost_usd": round(data["cost"], 2),
                "requests": data["requests"],
                "cache_hits": data["cache_hits"],
                "cache_rate": round(data["cache_hits"] / data["requests"] * 100, 1)
            }
            for day, data in sorted(daily.items())
        }

    def optimization_potential(self) -> Dict:
        """Calcula potencial de optimización."""
        summary = self.tracker.summary()
        current_cost = summary["total_cost_usd"]

        # Estimaciones de ahorro
        routing_savings = current_cost * 0.4  # Routing puede ahorrar 40%
        cache_savings = current_cost * 0.25  # Cache puede ahorrar 25%

        return {
            "current_cost": current_cost,
            "with_routing": current_cost - routing_savings,
            "with_cache": current_cost - cache_savings,
            "with_both": current_cost - (routing_savings + cache_savings),
            "monthly_projection": current_cost * 30,
            "monthly_with_optimizations": (current_cost - routing_savings - cache_savings) * 30
        }

# Uso
dashboard = CostDashboard(tracker)

print("=== Resumen Diario ===")
for day, data in dashboard.daily_summary().items():
    print(f"{day}: ${data['cost_usd']} ({data['requests']} requests, {data['cache_rate']}% cache)")

print("\n=== Potencial de Optimización ===")
opt = dashboard.optimization_projection()
print(f"Costo actual (mes): ${opt['monthly_projection']:.2f}")
print(f"Con routing + cache: ${opt['monthly_with_optimizations']:.2f}")
print(f"Ahorro potencial: ${opt['monthly_projection'] - opt['monthly_with_optimizations']:.2f}/mes")
