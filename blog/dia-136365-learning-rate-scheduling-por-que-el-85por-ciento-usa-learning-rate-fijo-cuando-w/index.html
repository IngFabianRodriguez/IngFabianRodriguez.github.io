Learning Rate Scheduling - Por qué el 85% usa learning rate fijo cuando warmup y decay son críticos

16 de april de 2026
torch.optim

Hola, parcero. Llegamos al Día 136 y hoy vamos con Learning Rate Scheduling. Este tema es crítico porque el 85% de los practitioners usan learning rate fijo cuando el scheduling adecuado puede ser la diferencia entre una pérdida que baja suave y una que diverge. Acompañame que esto va bien detallado, con código PyTorch real y los errores que más veo.

📐 Por Qué el Learning Rate Fijo No Es Óptimo

Con un learning rate fijo, tienes que elegir entre tres males:

  • LR muy alto: diverge o oscila alrededor del mínimo.
  • LR muy bajo: convergencia lenta, se estanca en platos.
  • LR intermedio: puede funcionar, pero nunca es óptimo para todas las fases del training.

La intuitición del scheduling es simple: al inicio del training, los gradientes son grandes y erráticos — necesitas un LR alto para explorar rápido. Hacia el final, el modelo está cerca del mínimo y necesitas LR bajo para hacer ajustes finos sin overshoot.

Matemáticamente, la loss surface cambia durante el training. Lo que es un buen LR en epoch 1 puede ser destructivo en epoch 50:

\theta_{t+1} = \theta_t - \eta_t \cdot \nabla_{\theta} L(\theta_t)

Con \eta_t variable obtenemos convergencia más rápida y mejor final loss.

class SimpleNet(nn.Module):
    """Red simple para experimento rápido."""
    def __init__(self, input_dim=10, hidden=64, output=1):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(input_dim, hidden),
            nn.ReLU(),
            nn.Linear(hidden, hidden),
            nn.ReLU(),
            nn.Linear(hidden, output),
        )

    def forward(self, x):
        return self.net(x)


def get_training_data(batch_size=64, n_samples=2000):
    """Datos sintéticos con patrón no-lineal."""
    X = torch.randn(n_samples, 10)
    # Función objetivo con interacciones no-lineales
    y = (X[:, 0] * X[:, 1] + X[:, 2] ** 2 - X[:, 3] * X[:, 4] +
         torch.sin(X[:, 5] + X[:, 6]) + 0.1 * torch.randn(n_samples))
    y = y.unsqueeze(1)
    return torch.utils.data.TensorDataset(X, y)


def plot_lr_schedules():
    """Visualiza todos los schedulers."""
    fig, axes = plt.subplots(2, 3, figsize=(14, 8))
    axes = axes.flatten()

    num_steps = 100
    initial_lr = 0.1

    schedulers_info = [
        ("StepLR (step=30, gamma=0.1)",
         StepLR(optim.SGD(torch.nn.Linear(1,1).parameters(), lr=initial_lr),
                step_size=30, gamma=0.1)),
        ("ExponentialLR (gamma=0.95)",
         ExponentialLR(optim.SGD(torch.nn.Linear(1,1).parameters(), lr=initial_lr),
                       gamma=0.95)),
        ("CosineAnnealingLR (T_max=100)",
         CosineAnnealingLR(optim.SGD(torch.nn.Linear(1,1).parameters(), lr=initial_lr),
                           T_max=num_steps)),
        ("CosineAnnealingWarmRestarts (T_0=20)",
         CosineAnnealingWarmRestarts(
             optim.SGD(torch.nn.Linear(1,1).parameters(), lr=initial_lr),
             T_0=20, T_mult=2)),
        ("OneCycleLR (max_lr=0.1)",
         OneCycleLR(optim.SGD(torch.nn.Linear(1,1).parameters(), lr=initial_lr),
                    max_lr=0.1, steps_per_epoch=1, epochs=num_steps)),
        ("CyclicLR (base=1e-4, max=0.1)",
         CyclicLR(optim.SGD(torch.nn.Linear(1,1).parameters(), lr=initial_lr),
                  base_lr=1e-4, max_lr=0.1, mode='triangular2',
                  step_size_up=20, step_size_down=30)),
    ]

    for ax, (name, scheduler) in zip(axes, schedulers_info):
        lrs = []
        optimizer = optim.SGD(torch.nn.Linear(1,1).parameters(), lr=initial_lr)
        # recreate scheduler with this optimizer
        if "StepLR" in name:
            sch = StepLR(optimizer, step_size=30, gamma=0.1)
        elif "ExponentialLR" in name:
            sch = ExponentialLR(optimizer, gamma=0.95)
        elif "CosineAnnealingWarmRestarts" in name:
            sch = CosineAnnealingWarmRestarts(optimizer, T_0=20, T_mult=2)
        elif "OneCycleLR" in name:
            sch = OneCycleLR(optimizer, max_lr=0.1, steps_per_epoch=1, epochs=num_steps)
        elif "CyclicLR" in name:
            sch = CyclicLR(optimizer, base_lr=1e-4, max_lr=0.1,
                           mode='triangular2', step_size_up=20, step_size_down=30)
        else:
            sch = CosineAnnealingLR(optimizer, T_max=num_steps)

        for _ in range(num_steps):
            lrs.append(optimizer.param_groups[0]['lr'])
            sch.step()

        ax.plot(lrs)
        ax.set_title(name, fontsize=10)
        ax.set_xlabel('Step')
        ax.set_ylabel('LR')
        ax.grid(True, alpha=0.3)

    plt.tight_layout()
    plt.savefig('/tmp/lr_schedulers_comparison.png', dpi=150)
    print("✅ Gráfico guardado en /tmp/lr_schedulers_comparison.png")
    plt.close()


def train_with_scheduler(scheduler_name: str, scheduler, num_epochs=50):
    """
    Entrena con un scheduler específico y retorna el historial de loss y LR.
    """
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    model = SimpleNet().to(device)
    criterion = nn.MSELoss()
    optimizer = optim.Adam(model.parameters(), lr=0.01)
    scheduler = scheduler(optimizer)

    dataset = get_training_data()
    loader = torch.utils.data.DataLoader(dataset, batch_size=64, shuffle=True)

    history = {'loss': [], 'lr': []}

    for epoch in range(num_epochs):
        epoch_loss = 0.0
        for X_batch, y_batch in loader:
            X_batch, y_batch = X_batch.to(device), y_batch.to(device)
            optimizer.zero_grad()
            pred = model(X_batch)
            loss = criterion(pred, y_batch)
            loss.backward()
            optimizer.step()
            epoch_loss += loss.item()

        scheduler.step()
        avg_loss = epoch_loss / len(loader)
        current_lr = optimizer.param_groups[0]['lr']
        history['loss'].append(avg_loss)
        history['lr'].append(current_lr)

    return history


def compare_all_schedulers():
    """
    Compara todos los schedulers en el mismo entrenamiento.
    """
    print("\n" + "="*60)
    print("COMPARING ALL LR SCHEDULERS")
    print("="*60)

    # Gráfico de schedulers
    plot_lr_schedules()

    num_epochs = 50
    results = {}

    scheduler_configs = [
        ("Fixed LR (baseline)", None),
        ("StepLR (step=20, gamma=0.5)", lambda opt: StepLR(opt, step_size=20, gamma=0.5)),
        ("CosineAnnealing (T_max=50)", lambda opt: CosineAnnealingLR(opt, T_max=num_epochs)),
        ("CosineAnnealingWarmRestarts (T0=10, Tmult=2)",
         lambda opt: CosineAnnealingWarmRestarts(opt, T_0=10, T_mult=2)),
        ("OneCycleLR (max_lr=0.02)", lambda opt: OneCycleLR(
            opt, max_lr=0.02, steps_per_epoch=31, epochs=num_epochs, pct_start=0.3)),
        ("CyclicLR (triangular2)", lambda opt: CyclicLR(
            opt, base_lr=1e-5, max_lr=0.01, mode='triangular2',
            step_size_up=15, step_size_down=35)),
    ]

    for name, scheduler_fn in scheduler_configs:
        model = SimpleNet()
        criterion = nn.MSELoss()
        optimizer = optim.Adam(model.parameters(), lr=0.01)
        scheduler = scheduler_fn(optimizer) if scheduler_fn else None

        dataset = get_training_data()
        loader = torch.utils.data.DataLoader(dataset, batch_size=64, shuffle=True)

        losses, lrs = [], []
        for epoch in range(num_epochs):
            epoch_loss = 0.0
            for X_batch, y_batch in loader:
                optimizer.zero_grad()
                pred = model(X_batch)
                loss = criterion(pred, y_batch)
                loss.backward()
                optimizer.step()
                epoch_loss += loss.item()
            if scheduler:
                scheduler.step()
            losses.append(epoch_loss / len(loader))
            lrs.append(optimizer.param_groups[0]['lr'])

        results[name] = {'loss': losses, 'lr': lrs}
        final_loss = losses[-1]
        print(f"  {name:45s} → final_loss={final_loss:.6f}, final_lr={lrs[-1]:.6f}")

    return results


# ──────────────────────────────────────────────
# 2. LR RANGE TEST: Cómo encontrar el LR óptimo
# ──────────────────────────────────────────────

def lr_range_test(model, loader, start_lr=1e-6, end_lr=1.0, num_iterations=100):
    """
    LR Range Test (Leslie Smith, 2018).
    Entrena por un número de iteraciones aumentando el LR exponencialmente.
    El LR óptimo está donde la loss empieza a subir sostenidamente.

    El punto donde la loss deja de descender y empieza a aumentar
    bruscamente indica el LR máximo recomendado.
    """
    print("\n" + "="*60)
    print("LR RANGE TEST — Encuentra LR óptimo")
    print("="*60)

    optimizer = optim.Adam(model.parameters(), lr=start_lr)
    criterion = nn.MSELoss()

    # Factor de incremento por iteración
    lr_factor = (end_lr / start_lr) ** (1.0 / num_iterations)

    losses = []
    lrs = []

    model.train()
    iteration = 0
    for batch_x, batch_y in loader:
        # Ajustar LR
        for param_group in optimizer.param_groups:
            param_group['lr'] = start_lr * (lr_factor ** iteration)

        optimizer.zero_grad()
        pred = model(batch_x)
        loss = criterion(pred, batch_y)
        loss.backward()
        optimizer.step()

        current_lr = optimizer.param_groups[0]['lr']
        losses.append(loss.item())
        lrs.append(current_lr)

        if iteration >= num_iterations:
            break
        iteration += 1

    # Encontrar rango óptimo
    # El LR óptimo está donde la loss es mínima (no donde empieza a subir)
    min_loss_idx = np.argmin(losses)
    optimal_lr = lrs[min_loss_idx]

    print(f"  LR mínimo: {start_lr:.2e}")
    print(f"  LR máximo: {end_lr:.2e}")
    print(f"  LR óptimo (donde loss mínima): {optimal_lr:.2e}")
    print(f"  Loss en LR óptimo: {losses[min_loss_idx]:.6f}")
    print(f"\n  Recomendación: usa max_lr = {optimal_lr:.2e} a {optimal_lr*10:.2e}")
    print(f"  con OneCycleLR o CosineAnnealing.")

    return lrs, losses, optimal_lr

# 3. ONE CYCLIC LR — Implementación detallada
# ──────────────────────────────────────────────

class OneCycleLRManual:
    """
    OneCycleLR manual para entender exactamente qué hace.
    Tres fases:
    1. Warmup: LR sube de min_lr a max_lr
    2. Annealing: LR baja de max_lr a min_lr con cosine
    """
    def __init__(self, optimizer, max_lr: float, min_lr: float,
                 num_steps: int, pct_warmup: float = 0.3):
        self.optimizer = optimizer
        self.max_lr = max_lr
        self.min_lr = min_lr
        self.num_steps = num_steps
        self.pct_warmup = pct_warmup
        self.warmup_steps = int(num_steps * pct_warmup)
        self.anneal_steps = num_steps - self.warmup_steps
        self.current_step = 0

    def step(self):
        self.current_step += 1
        if self.current_step <= self.warmup_steps:
            # Fase 1: warmup lineal
            lr = self.min_lr + (self.max_lr - self.min_lr) * (self.current_step / self.warmup_steps)
        else:
            # Fase 2: cosine annealing
            progress = (self.current_step - self.warmup_steps) / self.anneal_steps
            lr = self.max_lr - (self.max_lr - self.min_lr) * (0.5 * (1 + torch.cos(torch.tensor(progress * 3.14159)).item()))
        for param_group in self.optimizer.param_groups:
            param_group['lr'] = lr
        return lr

📊 Guía de Schedulers — ¿Cuándo Usar Cada Uno?

  • StepLR: Decaimiento por pasos. Simple pero rígido. Úsalo en modelos pequeños o cuando no tienes tiempo de afinar.
  • CosineAnnealingLR: Decaimiento suave con cosine. Excelente para cierre de training — típicamente mejora 5-15% sobre step decay.
  • CosineAnnealingWarmRestarts (SGDR): Like cosine pero con "rebotes". Ideal para salir de mínimos locales en训练的后期阶段.
  • OneCycleLR: El más robusto para la mayoría de casos. Un ciclo de LR que sube y baja. Típicamente produce los mejores resultados con menos tuning.
  • CyclicLR: LR oscila entre dos bounds. Útil para跳出局部最小值 pero requiere más tuning de bounds.

🚨 Los 3 Errores Más Comunes

❌ Error 1: No usar warmup en Transformers

Los Transformers tienen attention mechanisms que son extremamente sensibles a LR alto al inicio. Sin warmup, los primeros updates pueden causar loss spikes que arruinan el pre-training desde el día uno.

# ❌ MAL: LR alto desde el primer paso en un Transformer
optimizer = optim.Adam(model.parameters(), lr=0.001)
# Sin warmup → loss spikes en early training

# ✅ BIEN: warmup de 10% de las steps totales
scheduler = OneCycleLR(
    optimizer,
    max_lr=0.001,
    pct_start=0.1,  # 10% de steps para warmup
    steps_per_epoch=len(train_loader),
    epochs=num_epochs,
)
# OneCycleLR tiene warmup built-in con pct_start

❌ Error 2: Cosine Annealing sin min_lr adecuada — LR se muere

Many practitioners use cosine annealing with eta_min=0 (default). Cuando el LR llega a 0, el modelo deja de aprender en las últimas épocas. Esto puede ser un 20-30% de tu training desperdiciado.

# ❌ MAL: min_lr=0 — el LR muere completamente
scheduler = CosineAnnealingLR(optimizer, T_max=50, eta_min=0)

# ✅ BIEN: min_lr = 1/100 to 1/10 del LR inicial
scheduler = CosineAnnealingLR(optimizer, T_max=50, eta_min=1e-5)
# Mantén un LR pequeño pero有意义 hacia el final del training

# También puedes usar CosineAnnealingWarmRestarts para "revivir" el LR
scheduler = CosineAnnealingWarmRestarts(optimizer, T_0=20, T_mult=2, eta_min=1e-6)

❌ Error 3: OneCycleLR con max_lr muy alto — diverge

OneCycleLR asume que el max_lr es el LR máximo durante el ciclo. Many practitioners pick a value that's too high, especially when coming from a fixed LR baseline that was already working. OneCycleLR needs a properly tuned max_lr — not just "10x whatever you were using before".

# ❌ MAL: max_lr=0.1 cuando el LR óptimo es 0.001
# OneCycleLR will try to push through 0.1 → diverge
scheduler = OneCycleLR(optimizer, max_lr=0.1, ...)

# ✅ BIEN: primero corre lr_range_test para encontrar el LR óptimo
# Luego usa 2-10x el LR óptimo como max_lr de OneCycleLR
optimal_lr = 0.002  # del range test
scheduler = OneCycleLR(
    optimizer,
    max_lr=0.01,  # 5x el LR óptimo — suficientemente alto para explorar
    pct_start=0.3,  # 30% warmup
    div_factor=25,   # initial_lr = max_lr / 25 = 0.0004
    final_div_factor=100,  # final_lr = max_lr / 2500 = 0.000004
)

✅ Resumen — Día 136

  • LR fijo es suboptimal — el training tiene múltiples fases que requieren diferentes LR.
  • OneCycleLR es el más robusto para la mayoría de casos: un ciclo de warmup + cosine annealing.
  • CosineAnnealingWarmRestarts (SGDR) es excelente para训练的晚期 cuando necesitas escapar mínimos locales.
  • Siempre haz un LR Range Test antes de pick un scheduler — el LR óptimo varies por modelo y dataset.
  • Para Transformers: usa warmup (pct_start=0.1) — es no negociable.
  • No dejes min_lr=0 en cosine annealing — usa min_lr = initial_lr / 100.
  • OneCycleLR max_lr debe ser 2-10x el LR óptimo del range test, no un valor arbitrario.
