Intro - El Modelo Perfecto Que Nunca Llegó a Producción

Hace un mes un amigo me contó con orgullo que había entrenado un modelo de ML que predecía con 94% de accuracy la churning de clientes para una empresa de telecom en Medellín. "Es increíble", me dijo. Le pregunté: "¿Y dónde está corriendo?" Me miró con ojos de cordero: "¿Cómo que dónde está corriendo? Está en mi laptop, funcionando perfecto."

Ahí está el problema. El 85% de los modelos de ML nunca llegan a producción. Se quedan en notebooks, en laptops, en repositorios que nadie mira. Y el verdadero trabajo —deploy, monitoring, maintenance— es lo que el 85% ignora porque prefiere quedarse en la zona de confort de Jupyter.

Hoy vamos a hablar de "The Last Mile" de ML: cómo pasar del modelo en tu laptop a producción, cómo monitorear para detectar drift, cómo hacer A/B testing, y cómo el 15% que sí hace esto bien está dejando atrás al 85%.

Los componentes clave:

  • Model Server: Sirve predicciones con baja latencia
  • Feature Store: Gestiona features para training e inference
  • Model Registry: Versiona modelos y metadatos
  • Monitoring: Detecta drift, degrdación de performance
  • CI/CD: Automatiza deployment de nuevos modelos

from dataclasses import dataclass, field
from typing import List, Dict, Optional, Any, Callable
from datetime import datetime, timedelta
from enum import Enum
import pickle
import hashlib
import json
import time

class ModelStatus(Enum):
    """Estados del modelo en el lifecycle."""
    STAGING = "staging"
    PRODUCTION = "production"
    DEPRECATED = "deprecated"
    ROLLED_BACK = "rolled_back"

@dataclass
class ModelVersion:
    """Versión de un modelo con metadatos."""
    version_id: str
    model_path: str
    created_at: datetime
    metrics: Dict[str, float]
    features: List[str]
    status: ModelStatus
    creator: str
    description: str = ""
    parent_version: Optional[str] = None
    
    @property
    def version_hash(self) -> str:
        """Hash único de la versión."""
        content = f"{self.version_id}{self.created_at.isoformat()}"
        return hashlib.md5(content.encode()).hexdigest()[:8]

@dataclass
class PredictionRequest:
    """Request de predicción."""
    request_id: str
    features: Dict[str, Any]
    timestamp: datetime
    model_version: str
    metadata: Dict[str, Any] = field(default_factory=dict)

@dataclass
class PredictionResponse:
    """Response de predicción."""
    request_id: str
    prediction: Any
    model_version: str
    latency_ms: float
    timestamp: datetime
    features_used: List[str]

class FeatureStore:
    """Feature Store para gestión de features."""
    
    def __init__(self):
        self.features: Dict[str, List[float]] = {}
        self.feature_stats: Dict[str, Dict] = {}
    
    def register_feature(self, name: str, values: List[float]):
        """Registra una nueva feature."""
        self.features[name] = values
        self.feature_stats[name] = {
            "mean": sum(values) / len(values),
            "min": min(values),
            "max": max(values),
            "count": len(values),
            "std": self._std(values)
        }
    
    def _std(self, values: List[float]) -> float:
        """Calcula desviación estándar."""
        mean = sum(values) / len(values)
        variance = sum((x - mean) ** 2 for x in values) / len(values)
        return variance ** 0.5
    
    def get_feature_stats(self, name: str) -> Dict:
        """Obtiene estadísticas de una feature."""
        return self.feature_stats.get(name, {})
    
    def validate_features(self, features: Dict[str, Any]) -> bool:
        """Valida que features estén en el store."""
        for name in features.keys():
            if name not in self.features:
                return False
        return True

class ModelRegistry:
    """Registry de modelos con versionado."""
    
    def __init__(self, storage_path: str = "./models"):
        self.storage_path = storage_path
        self.models: Dict[str, ModelVersion] = {}
        self.production_model: Optional[str] = None
        self.staging_model: Optional[str] = None
    
    def register_model(
        self,
        model: Any,
        version_id: str,
        metrics: Dict[str, float],
        features: List[str],
        creator: str,
        description: str = ""
    ) -> ModelVersion:
        """Registra un nuevo modelo."""
        
        # Guardar modelo en disco
        model_path = f"{self.storage_path}/{version_id}.pkl"
        with open(model_path, 'wb') as f:
            pickle.dump(model, f)
        
        # Crear versión
        model_version = ModelVersion(
            version_id=version_id,
            model_path=model_path,
            created_at=datetime.now(),
            metrics=metrics,
            features=features,
            status=ModelStatus.STAGING,
            creator=creator,
            description=description,
            parent_version=self.production_model
        )
        
        self.models[version_id] = model_version
        self.staging_model = version_id
        
        return model_version
    
    def promote_to_production(self, version_id: str) -> bool:
        """Promueve un modelo de staging a production."""
        
        if version_id not in self.models:
            return False
        
        # Despromover producción actual
        if self.production_model:
            self.models[self.production_model].status = ModelStatus.DEPRECATED
        
        # Promover nuevo
        self.models[version_id].status = ModelStatus.PRODUCTION
        self.production_model = version_id
        self.staging_model = None
        
        return True
    
    def rollback(self) -> bool:
        """Rollback al modelo anterior."""
        
        if not self.production_model:
            return False
        
        current = self.models[self.production_model]
        if current.parent_version and current.parent_version in self.models:
            # Despromover actual
            current.status = ModelStatus.ROLLED_BACK
            # Restaurar padre
            self.models[current.parent_version].status = ModelStatus.PRODUCTION
            self.production_model = current.parent_version
            return True
        
        return False
    
    def get_production_model(self) -> Optional[ModelVersion]:
        """Obtiene el modelo en producción."""
        if self.production_model:
            return self.models[self.production_model]
        return None

class DriftDetector:
    """Detector de drift en features y predicciones."""
    
    def __init__(self, baseline_stats: Dict[str, Dict]):
        self.baseline_stats = baseline_stats
        self.current_window: List[Dict[str, float]] = []
        self.window_size = 1000
    
    def add_sample(self, features: Dict[str, float], prediction: float):
        """Agrega una muestra al window."""
        sample = {"prediction": prediction, **features}
        self.current_window.append(sample)
        
        if len(self.current_window) > self.window_size:
            self.current_window.pop(0)
    
    def compute_drift(self) -> Dict[str, float]:
        """Calcula métricas de drift."""
        
        if len(self.current_window) < 100:
            return {"status": "insufficient_data"}
        
        # Calcular estadísticas actuales
        current_stats = {}
        for key in self.baseline_stats.keys():
            values = [s.get(key, 0) for s in self.current_window]
            if values:
                current_stats[key] = {
                    "mean": sum(values) / len(values),
                    "std": self._std(values)
                }
        
        # Calcular drift (PSI - Population Stability Index)
        drift_scores = {}
        for key in self.baseline_stats.keys():
            if key in current_stats:
                baseline = self.baseline_stats[key]
                current = current_stats[key]
                
                # Simple drift: diferencia relativa de medias
                if baseline["mean"] != 0:
                    drift = abs(current["mean"] - baseline["mean"]) / abs(baseline["mean"])
                    drift_scores[key] = drift
        
        # PSI para predicción
        if self.current_window:
            preds = [s["prediction"] for s in self.current_window]
            pred_mean = sum(preds) / len(preds)
            baseline_pred = self.baseline_stats.get("prediction", {}).get("mean", 0)
            if baseline_pred != 0:
                drift_scores["prediction_drift"] = abs(pred_mean - baseline_pred) / abs(baseline_pred)
        
        return {
            "status": "ok" if all(d < 0.1 for d in drift_scores.values()) else "drift_detected",
            "scores": drift_scores,
            "window_size": len(self.current_window)
        }
    
    def _std(self, values: List[float]) -> float:
        """Calcula desviación estándar."""
        if not values:
            return 0.0
        mean = sum(values) / len(values)
        variance = sum((x - mean) ** 2 for x in values) / len(values)
        return variance ** 0.5

class MLServer:
    """Server de ML para producción."""
    
    def __init__(
        self,
        model_registry: ModelRegistry,
        feature_store: FeatureStore,
        drift_detector: DriftDetector
    ):
        self.registry = model_registry
        self.feature_store = feature_store
        self.drift_detector = drift_detector
        self.model = None
        self._load_production_model()
    
    def _load_production_model(self):
        """Carga el modelo de producción."""
        prod_model = self.registry.get_production_model()
        if prod_model:
            with open(prod_model.model_path, 'rb') as f:
                self.model = pickle.load(f)
            print(f"Loaded model version: {prod_model.version_id}")
    
    def predict(self, features: Dict[str, Any]) -> PredictionResponse:
        """Hace una predicción."""
        
        start_time = time.time()
        
        # Validar features
        if not self.feature_store.validate_features(features):
            raise ValueError("Features inválidas o faltantes")
        
        # Hacer predicción
        model_version = self.registry.get_production_model()
        if not self.model:
            raise RuntimeError("No hay modelo en producción")
        
        # Asumiendo que self.model tiene método predict
        prediction = self.model.predict([list(features.values())])[0]
        
        # Calcular latencia
        latency_ms = (time.time() - start_time) * 1000
        
        # Agregar al detector de drift
        self.drift_detector.add_sample(features, prediction)
        
        return PredictionResponse(
            request_id="",  # Llenar con ID real
            prediction=prediction,
            model_version=model_version.version_id if model_version else "unknown",
            latency_ms=latency_ms,
            timestamp=datetime.now(),
            features_used=list(features.keys())
        )

Los 3 Errores Que el 85% Comete

Error 1: No Tener Feature Store

El error más común. La gente entrena con features bonitas en pandas, y luego en producción las calcula diferente — o no las calcula du all. Training-serving skew es el asesino silencioso de modelos en producción.

¿Por qué está mal? Si las features que usas en training son diferentes de las que usas en inference, tu modelo está prediciendo con basura. Garbage in, garbage out — pero de forma sutil y difícil de detectar.

La solución: Usa un feature store (Feast, Tecton, o built-in) que sirva las mismas features para training e inference.

class SimpleFeatureStore:
    """Feature Store simple para evitar training-serving skew."""
    
    def __init__(self):
        self.features: Dict[str, Callable] = {}
        self.cache: Dict[str, Any] = {}
    
    def register_feature(self, name: str, calculator: Callable[[], Any]):
        """Registra una feature con su función calculadora.
        
        La misma función se usa para training e inference.
        """
        self.features[name] = calculator
    
    def get_features(self, feature_names: List[str]) -> Dict[str, Any]:
        """Obtiene features calculadas consistentemente."""
        return {name: self.features[name]() for name in feature_names}
    
    def get_training_features(self, feature_names: List[str], historical_data: List) -> List[Dict]:
        """Obtiene features para múltiples registros históricos.
        
        USA LA MISMA FUNCIÓN que inference — sin skew.
        """
        results = []
        for record in historical_data:
            # Aplicar a cada registro histórico
            # (En implementación real, usarías entity dataframe)
            feats = {name: self.features[name]() for name in feature_names}
            results.append(feats)
        return results

# Ejemplo de skew
def demo_skew_problem():
    """
    Problema: feature calculada diferente en training vs inference.
    
    Training (offline):  age = (current_year - birth_year)
    Inference (online):   age = user_profile['age']
    
    Si user_profile no se actualiza, el age diverge.
    """
    
    feature_store = SimpleFeatureStore()
    
    def calculate_age(birth_year: int) -> int:
        """Misma función para training e inference."""
        return 2026 - birth_year
    
    feature_store.register_feature('age', lambda: calculate_age(1990))
    
    # Training: usa la función
    training_age = feature_store.get_features(['age'])
    
    # Inference: SIEMPRE usa la misma función
    inference_age = feature_store.get_features(['age'])
    
    print(f"Training age: {training_age}")
    print(f"Inference age: {inference_age}")
    print(f"No skew: {training_age == inference_age}")

Error 2: No Monitorear Drift

El segundo error crítico. Una vez deployado, el modelo se queda ahí funcionando sin que nadie lo mire. Pero el mundo cambia, los datos cambian, y de repente tu modelo de 94% de accuracy está prediciendo con 67%.

¿Por qué está mal? El data drift y concept drift son reales. Los patrones de hace 6 meses no son los de hoy. Sin monitoreo, no sabes cuándo tu modelo está empezando a fallar.

La solución: Implementa monitoreo continuo de drift con alertas.

class ProductionMonitor:
    """Monitor de producción con alertas."""
    
    def __init__(
        self,
        drift_detector: DriftDetector,
        alert_threshold: float = 0.1,
        degradation_threshold: float = 0.05
    ):
        self.drift_detector = drift_detector
        self.alert_threshold = alert_threshold
        self.degradation_threshold = degradation_threshold
        self.alerts: List[Dict] = []
        self.metrics_history: List[Dict] = []
    
    def check_health(self) -> Dict[str, Any]:
        """Check de salud del sistema."""
        
        drift_result = self.drift_detector.compute_drift()
        
        health_status = {
            "timestamp": datetime.now().isoformat(),
            "drift_status": drift_result["status"],
            "drift_scores": drift_result.get("scores", {}),
            "alerts_triggered": []
        }
        
        # Check de drift
        if drift_result["status"] == "drift_detected":
            for feature, score in drift_result["scores"].items():
                if score > self.alert_threshold:
                    alert = {
                        "type": "drift",
                        "feature": feature,
                        "score": score,
                        "threshold": self.alert_threshold,
                        "severity": "high" if score > self.alert_threshold * 2 else "medium"
                    }
                    health_status["alerts_triggered"].append(alert)
                    self.alerts.append(alert)
        
        # Guardar en historial
        self.metrics_history.append(health_status)
        
        return health_status
    
    def get_performance_report(self) -> str:
        """Genera reporte de performance."""
        
        if not self.metrics_history:
            return "No hay datos de métricas todavía."
        
        recent = self.metrics_history[-100:]
        drift_count = sum(1 for m in recent if m["drift_status"] == "drift_detected")
        alert_count = sum(len(m.get("alerts_triggered", [])) for m in recent)
        
        return f"""
# Production Health Report

**Período:** Últimas {len(recent)} samples
**Fecha:** {datetime.now().isoformat()}

## Drift
- Muestras con drift detectado: {drift_count} ({drift_count/len(recent)*100:.1f}%)
- Alertas totales: {alert_count}

## Alertas Recientes
"""
    
    def should_retrain(self) -> bool:
        """Determina si es necesario re-entrenar."""
        
        if len(self.metrics_history) < 100:
            return False
        
        recent = self.metrics_history[-100:]
        drift_count = sum(1 for m in recent if m["drift_status"] == "drift_detected")
        
        # Re-entrenar si más del 30% de samples recientes tienen drift
        return (drift_count / len(recent)) > 0.3

Error 3: No Hacer A/B Testing de Modelos

El tercer error: desplegar un nuevo modelo y esperar que sea mejor sin verificarlo. O peor — no tener forma de comparar el modelo nuevo vs el actual.

¿Por qué está mal? Puedes desplegar un modelo que parece mejor en validation pero en production es peor. Sin A/B testing, no tienes forma de saberlo.

La solución: Implementa un framework de A/B testing con traffic splitting.

import random
from typing import Callable, Dict, List, Tuple
from dataclasses import dataclass

@dataclass
class ABTestConfig:
    """Configuración de un A/B test."""
    test_id: str
    control_version: str
    treatment_version: str
    traffic_percentage: float  # % de tráfico para treatment
    start_date: datetime
    end_date: Optional[datetime] = None
    min_samples: int = 1000

class ABTestManager:
    """Gestor de A/B tests para modelos."""
    
    def __init__(self):
        self.tests: Dict[str, ABTestConfig] = {}
        self.results: Dict[str, Dict] = {}
    
    def create_test(
        self,
        test_id: str,
        control_version: str,
        treatment_version: str,
        traffic_percentage: float = 0.1
    ) -> ABTestConfig:
        """Crea un nuevo A/B test."""
        
        config = ABTestConfig(
            test_id=test_id,
            control_version=control_version,
            treatment_version=treatment_version,
            traffic_percentage=traffic_percentage,
            start_date=datetime.now()
        )
        
        self.tests[test_id] = config
        self.results[test_id] = {
            "control": {"predictions": [], "labels": []},
            "treatment": {"predictions": [], "labels": []}
        }
        
        return config
    
    def get_model_for_request(self, test_id: str) -> str:
        """Determina qué modelo usar para un request.
        
        Returns 'control', 'treatment', o None si no hay test activo.
        """
        
        if test_id not in self.tests:
            return None
        
        test = self.tests[test_id]
        
        # Check si el test está activo
        if test.end_date and datetime.now() > test.end_date:
            return None
        
        # Traffic splitting
        if random.random() < test.traffic_percentage:
            return "treatment"
        return "control"
    
    def record_result(
        self,
        test_id: str,
        variant: str,
        prediction: float,
        label: Optional[float] = None
    ):
        """Registra resultado de predicción."""
        
        if test_id not in self.results:
            return
        
        self.results[test_id][variant]["predictions"].append(prediction)
        if label is not None:
            self.results[test_id][variant]["labels"].append(label)
    
    def evaluate_test(self, test_id: str) -> Dict:
        """Evalúa resultado de un A/B test."""
        
        if test_id not in self.tests or test_id not in self.results:
            return {"status": "test_not_found"}
        
        test = self.tests[test_id]
        res = self.results[test_id]
        
        control_preds = res["control"]["predictions"]
        treatment_preds = res["treatment"]["predictions"]
        
        # Verificar que tenemos suficientes samples
        if len(control_preds) < test.min_samples or len(treatment_preds) < test.min_samples:
            return {
                "status": "insufficient_samples",
                "control_n": len(control_preds),
                "treatment_n": len(treatment_preds),
                "min_required": test.min_samples
            }
        
        # Calcular métricas
        control_mean = sum(control_preds) / len(control_preds)
        treatment_mean = sum(treatment_preds) / len(treatment_preds)
        
        # Si tenemos labels, calcular accuracy/error
        control_labels = res["control"]["labels"]
        treatment_labels = res["treatment"]["labels"]
        
        metrics = {
            "status": "complete",
            "test_id": test_id,
            "control_version": test.control_version,
            "treatment_version": test.treatment_version,
            "control_n": len(control_preds),
            "treatment_n": len(treatment_preds),
            "control_mean_prediction": control_mean,
            "treatment_mean_prediction": treatment_mean,
            "improvement": (treatment_mean - control_mean) / control_mean if control_mean != 0 else 0
        }
        
        if control_labels and treatment_labels:
            # Calcular error si hay labels
            control_error = sum((p - l) ** 2 for p, l in zip(control_preds, control_labels)) / len(control_labels)
            treatment_error = sum((p - l) ** 2 for p, l in zip(treatment_preds, treatment_labels)) / len(treatment_labels)
            
            metrics["control_mse"] = control_error
            metrics["treatment_mse"] = treatment_error
            metrics["mse_improvement"] = (control_error - treatment_error) / control_error if control_error != 0 else 0
            metrics["winner"] = "treatment" if treatment_error < control_error else "control"
        
        return metrics

def demo_ab_testing():
    """Demuestra A/B testing."""
    
    manager = ABTestManager()
    
    # Crear test: nuevo modelo vs actual
    test = manager.create_test(
        test_id="churn_model_v2",
        control_version="v1.0.0",
        treatment_version="v2.0.0",
        traffic_percentage=0.1  # 10% del tráfico va al nuevo modelo
    )
    
    print(f"Created test: {test.test_id}")
    print(f"Control: {test.control_version}")
    print(f"Treatment: {test.treatment_version}")
    print(f"Traffic: {test.traffic_percentage * 100}%")
    
    # Simular requests
    for i in range(10000):
        variant = manager.get_model_for_request("churn_model_v2")
        if variant:
            # Simular predicción
            prediction = 0.85 if variant == "control" else 0.87
            # Simular label (solo para algunas)
            label = 0.8 if i % 10 == 0 else None
            manager.record_result("churn_model_v2", variant, prediction, label)
    
    # Evaluar test
    result = manager.evaluate_test("churn_model_v2")
    print(f"\nTest Result: {result}")

Cierre - Colombianito

Entonces, ¿qué aprendimos hoy? Que el verdadero trabajo de ML no es entrenar el modelo — es mantenerlo vivo y funcionando en producción. Feature stores, drift detection, A/B testing, monitoring — todo esto es lo que separa a los que dicen "tenemos ML en producción" de los que de verdad lo tienen.

La próxima vez que alguien te diga que tiene un modelo de 94% de accuracy, pregúntale: ¿Cuánto hace que no actualiza los features? ¿Hay drift detection? ¿Puedes hacer rollback en 5 minutos si algo falla? Si la respuesta a cualquiera de estas es "no sé", entonces tiene un modelo de notebook, no un modelo de producción.

En resumen: si tu modelo no está en producción con monitoreo, no tienes nada. Y si está en producción sin monitoring, tienes un liability, no un asset. Pasa del 85% que tiene notebooks al 15% que tiene sistemas de producción de verdad.

Y eso, mi gente, es lo que separa a los data scientists de los ML engineers de verdad.
