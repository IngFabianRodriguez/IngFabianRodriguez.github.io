¿Qué es Image Captioning?

Image Captioning es la tarea de generar una descripción textual automática para una imagen. Combina visión por computador y procesamiento de lenguaje natural para producir oraciones que describan el contenido visual de una imagen.

Aplicaciones prácticas incluyen:

  • Accesibilidad: Describir imágenes para personas con discapacidad visual
  • Indexación visual: Generar texto para búsqueda de imágenes
  • Redes sociales: Auto-generar captions para fotos
  • Documentación médica: Describir radiografías y scans

Arquitectura Encoder-Decoder

El enfoque estándar combina una CNN (encoder) con un RNN/LSTM (decoder):

"""
Image Captioning con CNN-Encoder y LSTM-Decoder
Arquitectura encoder-decoder clásica para captioning
"""
import torch
import torch.nn as nn
import torchvision.models as models
from dataclasses import dataclass
from typing import List, Optional

@dataclass
class CaptionResult:
    caption: str
    probabilities: List[float]
    attention_weights: Optional[torch.Tensor] = None

class ImageCaptioner(nn.Module):
    """
    CNN Encoder + LSTM Decoder para Image Captioning
    """
    
    def __init__(
        self,
        embed_size: int = 256,
        hidden_size: int = 512,
        vocab_size: int = 10000,
        encoder_dim: int = 2048,
        dropout: float = 0.3
    ):
        super().__init__()
        
        # Encoder: CNN pre-entrenada (ResNet)
        self.encoder = models.resnet101(weights=models.ResNet101_Weights.DEFAULT)
        # Remover la última capa fully connected
        self.encoder = nn.Sequential(*list(self.encoder.children())[:-1])
        encoder_dim = 2048  # ResNet101 output dimension
        
        # Proyección de features de imagen al espacio de embedding
        self.embed = nn.Linear(encoder_dim, embed_size)
        
        #Embedding para palabras
        self.word_embedding = nn.Embedding(vocab_size, embed_size)
        
        # LSTM Decoder
        self.lstm = nn.LSTMCell(embed_size + embed_size, hidden_size)
        
        # Atencion visual (si se usa)
        self.attention = None  # Opcional: Bahdanau attention
        
        # Clasificador para predecir siguiente palabra
        self.fc = nn.Linear(hidden_size, vocab_size)
        
        self.dropout = nn.Dropout(dropout)
        self.hidden_size = hidden_size
        
        # Inicializacion de hidden state
        self.init_h = nn.Linear(encoder_dim, hidden_size)
        self.init_c = nn.Linear(encoder_dim, hidden_size)
        
    def encode_image(self, images: torch.Tensor) -> torch.Tensor:
        """Extrae features de la imagen usando CNN"""
        features = self.encoder(images)  # [B, 2048, 7, 7] para ResNet
        features = features.permute(0, 2, 3, 1)  # [B, 7, 7, 2048]
        batch_size = features.size(0)
        features = features.view(batch_size, -1, features.size(-1))  # [B, 49, 2048]
        return features
    
    def init_hidden_state(self, features: torch.Tensor) -> tuple:
        """Inicializa hidden y cell states del LSTM"""
        mean_features = features.mean(dim=1)  # [B, 2048]
        h = torch.tanh(self.init_h(mean_features))  # [B, hidden_size]
        c = torch.tanh(self.init_c(mean_features))  # [B, hidden_size]
        return h, c
    
    def forward(
        self,
        features: torch.Tensor,
        captions: torch.Tensor,
        caption_lengths: List[int]
    ) -> torch.Tensor:
        """
        Forward pass durante entrenamiento
        
        Args:
            features: Image features [B, num_pixels, encoder_dim]
            captions: Word indices [B, max_caption_length]
            caption_lengths: Longitudes reales de captions (sin padding)
        
        Returns:
            logits: Predicciones para cada posicion [B, max_len, vocab_size]
        """
        batch_size = features.size(0)
        device = features.device
        
        # Embedding de imagenes
        features = self.embed(features)  # [B, num_pixels, embed_size]
        
        # Embedding de captions (sin el ultimo token)
        embeddings = self.word_embedding(captions)  # [B, max_len, embed_size]
        
        # Concatenar image features al inicio (como primer input)
        h, c = self.init_hidden_state(features)
        
        # Desemcapsular para procesar secuencialmente
        predictions = []
        max_len = int(max(caption_lengths))
        
        for t in range(max_len):
            # Obtener embedding de la palabra actual
            word_embed = embeddings[:, t, :]  # [B, embed_size]
            
            # Combinar con context de imagen (promedio)
            context = features.mean(dim=1)  # [B, embed_size]
            
            # Concatenar word embedding + image context
            lstm_input = torch.cat([word_embed, context], dim=1)  # [B, embed_size*2]
            
            # LSTM step
            h, c = self.lstm(lstm_input, (h, c))
            
            # Prediccion de siguiente palabra
            preds = self.fc(self.dropout(h))  # [B, vocab_size]
            predictions.append(preds)
        
        # Stack predictions
        predictions = torch.stack(predictions, dim=1)  # [B, max_len, vocab_size]
        
        return predictions
    
    def generate_caption(
        self,
        features: torch.Tensor,
        vocab,
        max_length: int = 50,
        temperature: float = 1.0
    ) -> CaptionResult:
        """
        Genera caption para una imagen (inference)
        
        Args:
            features: Image features [1, num_pixels, encoder_dim]
            vocab: Vocabulario para decodificar
            max_length: Maxima longitud del caption
            temperature: Temperatura para sampling (1.0 = greedy)
        
        Returns:
            CaptionResult con caption generado y probabilidades
        """
        self.eval()
        device = features.device
        
        features = self.embed(features)
        h, c = self.init_hidden_state(features)
        
        # Iniciar con token START
        current_word = torch.tensor([[vocab.stoi[""]]], device=device)
        caption_ids = [current_word.item()]
        log_probs_list = []
        
        with torch.no_grad():
            for _ in range(max_length):
                embeddings = self.word_embedding(current_word)  # [1, embed_size]
                context = features.mean(dim=1)  # [1, embed_size]
                
                lstm_input = torch.cat([embeddings.squeeze(1), context], dim=1)
                h, c = self.lstm(lstm_input, (h, c))
                
                logits = self.fc(h)  # [1, vocab_size]
                
                # Apply temperature
                logits = logits / temperature
                probs = torch.softmax(logits, dim=-1)
                
                # Sample o greedy
                if temperature == 0:
                    next_word_idx = logits.argmax(dim=-1)
                else:
                    next_word_idx = torch.multinomial(probs, 1)
                
                next_word_idx = next_word_idx.item()
                caption_ids.append(next_word_idx)
                log_probs_list.append(probs[0, next_word_idx].item())
                
                # Stop si llega a END token
                if next_word_idx == vocab.stoi[""]:
                    break
                
                current_word = torch.tensor([[next_word_idx]], device=device)
        
        # Convertir indices a palabras
        caption_words = [vocab.itos[idx] for idx in caption_ids]
        caption = " ".join(caption_words[1:-1])  # Remover start y end
        
        return CaptionResult(
            caption=caption,
            probabilities=log_probs_list
        )

Evaluación de modelos de Captioning

Las metricas standard para evaluar captions incluyen BLEU, METEOR, ROUGE, CIDEr, y SPICE:

"""
Evaluacion de Image Captioning
Metricas: BLEU, METEOR, ROUGE-L, CIDEr, SPICE
"""
import torch
from typing import List, Dict
from collections import Counter
import numpy as np

class CaptionMetrics:
    """Calcula metricas de evaluacion para captions generados"""
    
    def __init__(self):
        self.reset()
    
    def reset(self):
        self.references = []
        self.hypotheses = []
    
    def update(self, hypothesis: str, references: List[str]):
        """Agrega un par de hipotesis y referencias para evaluar"""
        self.hypotheses.append(hypothesis.lower().split())
        self.references.append([r.lower().split() for r in references])
    
    def compute_bleu(self, n: int = 4) -> float:
        """BLEU score (1-4 gram precision con brevity penalty)"""
        from collections import Counter
        
        def modified_precision(references, hypothesis, n):
            hyp_ngrams = Counter(self._get_ngrams(hypothesis, n))
            
            max_counts = Counter()
            for ref in references:
                ref_ngrams = Counter(self._get_ngrams(ref, n))
                for ngram in hyp_ngrams:
                    max_counts[ngram] = max(max_counts.get(ngram, 0), ref_ngrams.get(ngram, 0))
            
            clipped_counts = min(hyp_ngrams[ngram], max_counts[ngram]) for ngram in hyp_ngrams
            
            return sum(clipped_counts) / max(sum(hyp_ngrams.values()), 1)
        
        def brevity_penalty(references, hypothesis):
            ref_len = min(len(r) for r in references)
            hyp_len = len(hypothesis)
            
            if hyp_len > ref_len:
                return 1.0
            return np.exp(1 - ref_len / hyp_len) if hyp_len > 0 else 0
        
        precisions = [modified_precision(self.references[i], self.hypotheses[i], n) 
                      for i in range(len(self.hypotheses))]
        
        # Media geometrica de precisions
        if precisions:
            geo_mean = np.exp(np.mean([np.log(p + 1e-10) for p in precisions]))
        else:
            geo_mean = 0
        
        penalty = np.mean([brevity_penalty(self.references[i], self.hypotheses[i]) 
                          for i in range(len(self.hypotheses))])
        
        return geo_mean * penalty
    
    def compute_cider(self) -> float:
        """CIDEr: Consensus-based Image Description Evaluation"""
        def cosine_similarity(references, hypothesis):
            ref_vec = self._tf_idf(references)
            hyp_vec = self._tf_idf([hypothesis])
            
            dot = sum(ref_vec.get(w, 0) * hyp_vec.get(w, 0) for w in set(ref_vec) | set(hyp_vec))
            norm_ref = np.sqrt(sum(v**2 for v in ref_vec.values()))
            norm_hyp = np.sqrt(sum(v**2 for v in hyp_vec.values()))
            
            return dot / (norm_ref * norm_hyp + 1e-10)
        
        scores = [cosine_similarity(self.references[i], self.hypotheses[i]) 
                  for i in range(len(self.hypotheses))]
        return np.mean(scores)
    
    def compute_spice(self) -> float:
        """
        SPICE: Semantic Propositional Image Caption Evaluation
        Usa scene graphs para comparar contenido semantico
        """
        # En implementacion real se usariaSPICE toolkit
        # Simulacion:
        return 0.45
    
    def compute_all(self) -> Dict[str, float]:
        """Calcula todas las metricas"""
        return {
            "BLEU-1": self.compute_bleu(1),
            "BLEU-2": self.compute_bleu(2),
            "BLEU-3": self.compute_bleu(3),
            "BLEU-4": self.compute_bleu(4),
            "CIDEr": self.compute_cider(),
            "SPICE": self.compute_spice()
        }
    
    @staticmethod
    def _get_ngrams(tokens: List[str], n: int) -> List[tuple]:
        """Genera n-grams desde una lista de tokens"""
        return tuple(tokens[i:i+n] for i in range(len(tokens) - n + 1))
    
    @staticmethod
    def _tf_idf(references: List[List[str]]) -> Counter:
        """Calcula TF-IDF para un conjunto de referencias"""
        counter = Counter()
        for ref in references:
            counter.update(set(ref))
        return counter
