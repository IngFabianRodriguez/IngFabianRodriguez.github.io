Clinical NLP: Entendiendo Notas Médicas

Los médicos escriben notas en lenguaje natural. Convertir esas notas en datos estructurados es un problema clásico de NLP. BERT y modelos especializados están revolucionando esto:

import torch
from transformers import AutoTokenizer, AutoModelForTokenClassification

class ClinicalBERT:
    def __init__(self, model_name="medicalai/ClinicalBERT"):
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        self.model = AutoModelForTokenClassification.from_pretrained(
            model_name,
            num_labels=9  # Entity types: DISEASE, DRUG, DOSAGE, etc.
        )
    
    def extract_entities(self, clinical_note: str) -> List[Dict]:
        """
        Extrae entidades médicas de notas clínicas.
        
        Entidades reconocidas:
        - CONDITION (enfermedades)
        - DRUG (medicamentos)
        - DOSAGE (dosis)
        - ROUTE (vía de administración)
        - FREQUENCY (frecuencia)
        - DURATION (duración)
        - SYMPTOM (síntomas)
        - PROCEDURE (procedimientos)
        - ANATOMY (anatomía)
        """
        inputs = self.tokenizer(
            clinical_note,
            return_tensors="pt",
            truncation=True,
            max_length=512
        )
        
        with torch.no_grad():
            outputs = self.model(**inputs)
        
        predictions = torch.argmax(outputs.logits, dim=2)
        tokens = self.tokenizer.convert_ids_to_tokens(
            inputs["input_ids"][0]
        )
        
        entities = self._decode_predictions(tokens, predictions[0])
        return entities
    
    def _decode_predictions(self, tokens, predictions):
        entities = []
        current_entity = []
        
        for token, pred_id in zip(tokens, predictions):
            label = self.model.config.id2label[pred_id.item()]
            
            if token in ["[CLS]", "[SEP]", "[PAD]"]:
                continue
            
            if label.startswith("B-"):
                if current_entity:
                    entities.append(current_entity)
                current_entity = {"type": label[2:], "text": token}
            elif label.startswith("I-") and current_entity:
                current_entity["text"] += " " + token
            else:
                if current_entity:
                    entities.append(current_entity)
                    current_entity = []
        
        if current_entity:
            entities.append(current_entity)
        
        return entities

# Arquitectura típica para medical imaging: UNet con attention

import torch
import torch.nn as nn

class AttentionGate(nn.Module):
    """Attention gates para focalizar en regiones relevantes."""
    def __init__(self, F_g, F_l, F_int):
        super().__init__()
        self.W_g = nn.Conv2d(F_g, F_int, 1)
        self.W_x = nn.Conv2d(F_l, F_int, 1)
        self.psi = nn.Conv2d(F_int, 1, 1)
        self.relu = nn.ReLU(inplace=True)
        self.sigmoid = nn.Sigmoid()
    
    def forward(self, g, x):
        g1 = self.W_g(g)
        x1 = self.W_x(x)
        psi = self.relu(g1 + x1)
        psi = self.sigmoid(self.psi(psi))
        return x * psi

class ResBlock(nn.Module):
    """Residual block con batch normalization."""
    def __init__(self, channels):
        super().__init__()
        self.conv1 = nn.Conv2d(channels, channels, 3, padding=1)
        self.bn1 = nn.BatchNorm2d(channels)
        self.conv2 = nn.Conv2d(channels, channels, 3, padding=1)
        self.bn2 = nn.BatchNorm2d(channels)
        self.relu = nn.ReLU(inplace=True)
    
    def forward(self, x):
        residual = x
        out = self.relu(self.bn1(self.conv1(x)))
        out = self.bn2(self.conv2(out))
        out += residual
        return self.relu(out)

Fraud Detection con LightGBM

Detectar fraude en tiempo real es crítica para cualquier sistema de payments. LightGBM es el gold standard por su velocidad y accuracy:

import lightgbm as lgb
import numpy as np
from sklearn.model_selection import train_test_split
import pandas as pd

class FraudDetector:
    """
    Detector de fraude usando LightGBM con features
    específicos para transacciones financieras.
    """
    
    def __init__(self):
        self.model = None
        self.feature_names = [
            'amount', 'hour_of_day', 'day_of_week',
            'merchant_category', 'card_present',
            'country_match', 'velocity_1h', 'velocity_24h',
            'avg_transaction_amount', 'num_cards_on_account',
            'is_high_risk_merchant', 'time_since_last_transaction',
            'distance_from_home', 'distance_from_last_transaction'
        ]
    
    def create_features(self, transaction: dict) -> np.ndarray:
        """Engineer features para una transacción."""
        features = {}
        
        # Features temporales
        features['hour_of_day'] = transaction['timestamp'].hour
        features['day_of_week'] = transaction['timestamp'].weekday()
        
        # Features deamount
        features['amount'] = transaction['amount']
        features['log_amount'] = np.log1p(transaction['amount'])
        
        # Features de velocidad (cuántas transacciones en ventana de tiempo)
        features['velocity_1h'] = self._calc_velocity(
            transaction['card_id'], window_minutes=60
        )
        features['velocity_24h'] = self._calc_velocity(
            transaction['card_id'], window_minutes=1440
        )
        
        # Features de ubicación
        if transaction.get('location'):
            features['distance_from_home'] = self._calc_distance(
                transaction['location'],
                transaction['home_location']
            )
        
        # Feature de merchant risk
        features['is_high_risk_merchant'] = int(
            transaction['merchant_id'] in self.high_risk_merchants
        )
        
        return np.array([features[name] for name in self.feature_names])
    
    def train(self, X_train, y_train, X_val, y_val):
        """Entrena el modelo con class imbalance handling."""
        
        # Calcular scale_pos_weight para class imbalance
        scale_pos_weight = np.sum(y_train == 0) / np.sum(y_train == 1)
        
        params = {
            'objective': 'binary',
            'metric': 'auc',
            'boosting_type': 'gbdt',
            'num_leaves': 31,
            'learning_rate': 0.05,
            'feature_fraction': 0.9,
            'bagging_fraction': 0.8,
            'bagging_freq': 5,
            'scale_pos_weight': scale_pos_weight,
            'verbose': -1
        }
        
        train_data = lgb.Dataset(X_train, label=y_train)
        val_data = lgb.Dataset(X_val, label=y_val, reference=train_data)
        
        self.model = lgb.train(
            params,
            train_data,
            num_boost_round=1000,
            valid_sets=[train_data, val_data],
            callbacks=[
                lgb.early_stopping(stopping_rounds=50),
                lgb.log_evaluation(100)
            ]
        )
        
        return self.model
    
    def predict(self, transaction: dict) -> dict:
        """Predice si una transacción es fraudulenta."""
        features = self.create_features(transaction)
        prob = self.model.predict(features.reshape(1, -1))[0]
        
        return {
            'is_fraud': prob > 0.5,
            'fraud_probability': prob,
            'action': 'approve' if prob < 0.3 else ('review' if prob < 0.7 else 'block')
        }

# Uso:

detector.train(X_train, y_train, X_val, y_val)

            market_data.get('order_book')
        )

        combined_signal = (
            0.4 * signals['technical'] +
            0.35 * signals['sentiment'] +
            0.25 * signals['microstructure']
        )
        
        return {
            'raw_signal': combined_signal,
            'position_size': self.risk_manager.calculate_position(
                combined_signal,
                market_data['volatility'].iloc[-1]
            ),
            'stop_loss': self.risk_manager.stop_loss,
            'take_profit': self.risk_manager.take_profit
        }
    
    def _technical_signal(self, data):
        # RSI
        delta = data['close'].diff()
        gain = (delta.where(delta > 0, 0)).rolling(14).mean()
        loss = (-delta.where(delta < 0, 0)).rolling(14).mean()
        rs = gain / loss
        rsi = 100 - (100 / (1 + rs))
        
        # MACD
        ema12 = data['close'].ewm(span=12).mean()
        ema26 = data['close'].ewm(span=26).mean()
        macd = ema12 - ema26
        signal_line = macd.ewm(span=9).mean()
        
        # Normalizar y combinar
        rsi_norm = (rsi - 50) / 50
        macd_norm = (macd - signal_line) / data['close'].std()
        
        return 0.6 * rsi_norm + 0.4 * macd_norm

Recommendation System con Collaborative Filtering

import numpy as np
from scipy.sparse import csr_matrix
from sklearn.neighbors import NearestNeighbors

class RecommenderSystem:
    """
    Hybrid recommender combinando:
    1. Collaborative Filtering (user-item interactions)
    2. Content-based (item features)
    3. Knowledge-based (rules)
    """
    
    def __init__(self, n_neighbors=20):
        self.n_neighbors = n_neighbors
        self.user_item_matrix = None
        self.item_features = None
        self.user_ids = None
        self.item_ids = None
        self.user_model = None
        self.item_model = None
    
    def fit(self, interactions_df: pd.DataFrame, items_df: pd.DataFrame):
        """
        interactions_df: user_id, item_id, rating, timestamp
        items_df: item_id, category, price, description_embedding
        """
        # Crear user-item matrix
        self.user_ids = interactions_df['user_id'].unique()
        self.item_ids = interactions_df['item_id'].unique()
        
        user_idx = {uid: i for i, uid in enumerate(self.user_ids)}
        item_idx = {iid: i for i, iid in enumerate(self.item_ids)}
        
        rows = interactions_df['user_id'].map(user_idx)
        cols = interactions_df['item_id'].map(item_idx)
        values = interactions_df['rating']
        
        self.user_item_matrix = csr_matrix(
            (values, (rows, cols)),
            shape=(len(self.user_ids), len(self.item_ids))
        )
        
        # Entrenar modelo de similitud de usuarios
        self.user_model = NearestNeighbors(
            metric='cosine',
            algorithm='brute',
            n_neighbors=min(self.n_neighbors, len(self.user_ids))
        )
        self.user_model.fit(self.user_item_matrix)
        
        # Guardar item features para content-based
        self.item_features = items_df.set_index('item_id')
        
        return self
    
    def recommend(self, user_id: int, n_items: int = 10) -> List[dict]:
        """Genera recomendaciones personalizadas."""
        
        if user_id not in self.user_ids:
            # Cold start: devolver items populares
            return self._popular_items(n_items)
        
        user_idx = self.user_ids.tolist().index(user_id)
        user_vector = self.user_item_matrix[user_idx]
        
        # Encontrar usuarios similares
        distances, indices = self.user_model.kneighbors(user_vector)
        
        # Weighted average de ratings de usuarios similares
        scores = np.zeros(self.user_item_matrix.shape[1])
        weights = np.zeros(self.user_item_matrix.shape[1])
        
        for dist, neighbor_idx in zip(distances[0], indices[0]):
            similarity = 1 - dist
            if similarity > 0:
                neighbor_vector = self.user_item_matrix[neighbor_idx].toarray().flatten()
                scores += similarity * neighbor_vector
                weights += similarity
        
        # Normalizar
        scores = np.divide(
            scores, weights,
            out=np.zeros_like(scores),
            where=weights != 0
        )
        
        # Excluir items ya valorados por el usuario
        user_ratings = self.user_item_matrix[user_idx].toarray().flatten()
        scores[user_ratings > 0] = -np.inf
        
        # Top N items
        top_item_indices = np.argsort(scores)[-n_items:][::-1]
        
        return [
            {
                'item_id': self.item_ids[idx],
                'score': scores[idx],
                'category': self.item_features.loc[self.item_ids[idx], 'category'],
                'price': self.item_features.loc[self.item_ids[idx], 'price']
            }
            for idx in top_item_indices
        ]
    
    def _popular_items(self, n):
        """Fallback para cold start."""
        total_ratings = np.array(
            self.user_item_matrix.sum(axis=0)
        ).flatten()
        top_indices = np.argsort(total_ratings)[-n:][::-1]
        
        return [
            {'item_id': self.item_ids[idx], 'score': total_ratings[idx]}
            for idx in top_indices
        ]

# Predicción de demanda para inventory management

class DemandForecaster:
    """Usa LightGBM con features de series temporales."""
    
    def create_time_features(self, df: pd.DataFrame) -> pd.DataFrame:
        df = df.copy()
        df['hour'] = df['timestamp'].dt.hour
        df['day_of_week'] = df['timestamp'].dt.dayofweek
        df['day_of_month'] = df['timestamp'].dt.day
        df['month'] = df['timestamp'].dt.month
        df['is_weekend'] = df['day_of_week'].isin([5, 6]).astype(int)
        df['is_month_start'] = df['timestamp'].dt.is_month_start.astype(int)
        df['is_month_end'] = df['timestamp'].dt.is_month_end.astype(int)
        
        # Lag features
        for lag in [1, 7, 14, 30]:
            df[f'demand_lag_{lag}'] = df['demand'].shift(lag)
        
        # Rolling statistics
        df['demand_rolling_mean_7'] = df['demand'].rolling(7).mean()
        df['demand_rolling_std_7'] = df['demand'].rolling(7).std()
        df['demand_rolling_mean_30'] = df['demand'].rolling(30).mean()
        
        return df.dropna()
    
    def predict(self, historical_data: pd.DataFrame, horizon_days: int = 30):
        """Predice demanda para los próximos N días."""
        features = self.create_time_features(historical_data)
        
        X = features[[c for c in features.columns if c not in ['demand', 'timestamp']]]
        y = features['demand']
        
        model = lgb.LGBMRegressor(n_estimators=500, learning_rate=0.05)
        model.fit(X, y)
        
        # Generar predicciones recursivamente
        last_data = historical_data.copy()
        predictions = []
        
        for day in range(horizon_days):
            features = self.create_time_features(last_data)
            X_next = features[[c for c in features.columns if c not in ['demand', 'timestamp']]].iloc[-1:]
            pred = model.predict(X_next)[0]
            
            predictions.append({'day': day + 1, 'predicted_demand': pred})
            
            # Agregar predicción para siguiente iteración
            new_row = last_data.iloc[-1:].copy()
            new_row['demand'] = pred
            last_data = pd.concat([last_data, new_row], ignore_index=True)
        
        return predictions

⚠️ Los 3 Errores Más Comunes (Y Sus Soluciones)

Error 1: Not Understanding Domain-Specific Constraints (HIPAA, PCI-DSS)

El problema: Implementas un modelo brillante en healthcare o finance sin entender las regulaciones. Resultado: multas millonarias o que el modelo nunca puede pasar a producción.

La realidad: HIPAA en healthcare exige encriptación, audit trails, y que ningún PHI (Protected Health Information) salga de sistemas controlados. PCI-DSS en payments tiene requisitos estrictos de manejo de datos de tarjetas.

La solución: Antes de escribir una línea de código, consulta con el equipo legal y compliance. Entiende:

  • Cuáles datos puedes usar para entrenar
  • Cómo deben almacenarse los datos
  • Qué auditorías necesitas pasar
  • Cómo se maneja el consentimiento del usuario

Error 2: Assuming the Same Approach Works Across Industries

El problema: "Ya sé hacer fraud detection, ahora voy a hacer medical diagnosis con el mismo enfoque." No funciona así.

La realidad: En fraud detection, recall del 99% puede ser overkill si genera demasiados false positives que frustran usuarios legítimos. En medical diagnosis, ese 1% de missed cancers puede costar vidas. Los costos de error son radicalmente diferentes.

La solución: Antes de empezar, define explícitamente:

# Define tu función de costo de negocio, no solo la métrica de ML

def business_cost_matrix(industry):
    if industry == "healthcare":
        # False negative es crítico
        return np.array([[0, 100],   # FP: bajo costo (review adicional)
                        [1000, 0]])  # FN: altísimo costo (cáncer no detectado)
    
    elif industry == "fraud":
        # Balance entre UX y seguridad
        return np.array([[0, 5],     # FP: fricción en UX
                        [50, 0]])    # FN: pérdida directa
    
    elif industry == "retail":
        # Costo de sobre-inventario vs. faltante
        return np.array([[10, 0],    # FP: costo de returns
                        [30, 0]])    # FN: pérdida de venta

Error 3: Ignoring Regulatory Requirements Específicos del Industry

El problema: Implementar un modelo que funciona perfectamente en test pero no puede pasar regulatory approval. Tiempo y dinero perdidos.

La realidad: FDA para medical devices, FCA para trading algorithms, GDPR para datos de europeos... cada industria tiene su framework regulatorio. El model transparency y explainability no son opcionales — son requerimentos.

La solución: Involucra reguladores temprano:

  • FDA (HealthTech): Usa predicciones como "auxiliar" no "autónomo", documenta todo
  • FCA (Finance): Algorithmic trading requiere backtesting extensivo y audit trails
  • GDPR: Derecho a explicación, consentimiento explícito, data minimization
