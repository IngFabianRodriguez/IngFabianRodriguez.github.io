Por Que Scale Cambia Todo

A pequeña escala, tu código de ML probablemente se ve algo asi:

model = load_model()
trainer = Trainer(model=model, data=train_data)
trainer.train()

A escala de producción con miles de GPUs, ese mismo código se convierte en:

# Esto no es pseudocodigo — esto es lo que realmente escribes
distribuir_modelo_a_8192_GPUs()
manejar_fallos_en_cientos_de_nodos()
sincronizar_gradientes_con_precision_mixta()
implementar_checkpointing_para_no_perder_3_dias_de_entrenamiento()
optimizar_communication_overlap_para_no_morir_de_hambre()

La diferencia entre 1 GPU y 8192 GPUs no es lineal. Es un cambio de paradigma completo.

Distributed Training con PyTorch DDP

PyTorch Distributed Data Parallel (DDP) es el estándar para entrenamiento distribuido. La idea básica:

  • Cada GPU tiene una copia del modelo
  • Cada GPU procesa un batch diferente
  • Al final de cada step, los gradientes se sincronizan (all-reduce)
  • Todos los modelos permanecen idénticos
import os
import torch
import torch.distributed as dist
from torch.nn.parallel import DistributedDataParallel as DDP
from torch.distributed.fsdp import FullyShardedDataParallel as FSDP

# Setup basico de DDP
def setup_distributed():
    # Variables de entorno que tu scheduler (SLURM/K8s) configura
    dist.init_process_group(
        backend="nccl",  # NVIDIA GPU-to-GPU
        init_method="env://",
        world_size=int(os.environ["WORLD_SIZE"]),
        rank=int(os.environ["RANK"])
    )
    
    # Configurar para que cada proceso vea solo su GPU
    local_rank = int(os.environ["LOCAL_RANK"])
    torch.cuda.set_device(local_rank)
    
    return local_rank

FSDP — Sharding de Modelo Entre GPUs

Cuando el modelo no cabe en una sola GPU, necesitas FSDP (Fully Sharded Data Parallel). Esto divide el modelo entre GPUs — cada GPU guarda solo su parte.

from torch.distributed.fsdp import FullyShardedDataParallel as FSDP
from torch.distributed.fsdp import ShardingStrategy, MixedPrecision
from torch.distributed.fsdp.wrap import transformer_auto_wrap_policy

# FSDP con sharding de parametros
def train_with_fsdp():
    setup_distributed()
    local_rank = int(os.environ["LOCAL_RANK"])
    
    # Modelo grande — 70B parametros no cabe en una GPU
    model = LLAMA70B()  # 140GB en float16
    
    # Auto wrap policy para transformers — envuelve cada capa transformer
    # en su propio FSDP wrapper
    auto_wrap_policy = functools.partial(
        transformer_auto_wrap_policy,
        transformer_layer_cls={
            TransformerEncoderLayer,
            TransformerDecoderLayer,
            LlamaDecoderLayer,
        }
    )
    
    # Mixed precision para reducir memoria
    mixed_precision_policy = MixedPrecision(
        param_dtype=torch.float16,
        reduce_dtype=torch.float16,
        buffer_dtype=torch.float16,
    )
    
    # Sharding strategy: FULL_SHARD divide params y grads entre GPUs
    model = FSDP(
        model,
        sharding_strategy=ShardingStrategy.FULL_SHARD,
        auto_wrap_policy=auto_wrap_policy,
        mixed_precision=mixed_precision_policy,
        device_id=local_rank,
        use_orig_params=True  # Importante para gradient checkpointing
    )
    
    # FSDP requiere steps de optimizacion diferentes
    # En lugar de optimizer.step() directo:
    # 1. Reducir gradientes entre shards
    # 2. Hacer optimizer step local
    # 3. Sincronizar si es necesario
    
    for batch in train_loader:
        outputs = model(batch)
        loss = compute_loss(outputs)
        
        # Backward — FSDP maneja el gradient sharding
        loss.backward()
        
        # No puedes hacer optimizer.step() directo con FSDP!
        # Necesitas:
        model.backward_step()
        model.optimizer_step()
        model.sync_buffers()

# Parametros de FSDP para 70B en 8xA100 80GB:
# Con FULL_SHARD: cada GPU guarda ~9GB de parametros + ~9GB de gradientes + ~9GB de optim states
# Total: ~27GB por GPU — cabe en 80GB

Fault Tolerance y Checkpointing

Con miles de GPUs, la estadística es implacable: si entrenas por 30 días, vas a tener fallos. La pregunta es cuánto pierdes cuando falla.

La Matemática del Fallo

Si tienes 1000 GPUs y cada una tiene 99.9% de uptime (muy bueno):

# Probabilidad de que al menos una GPU falle en un step
# Asumiendo fallos independientes

n_gpus = 1000
uptime_per_gpu = 0.999
steps_per_day = 1000

# Probabilidad de fallo en un step
p_fail_step = 1 - uptime_per_gpu

# Probabilidad de fallo en un día
p_fail_day = 1 - (uptime_per_gpu ** (steps_per_day * n_gpus))

# Con 1000 GPUs y 1000 steps/dia:
# p_fail_day ≈ 63% — vas a fallar casi todos los días!
# Con 10000 GPUs: p_fail_day ≈ 99.99%

print(f"Con 1000 GPUs: {p_fail_day*100:.1f}% probabilidad de fallo por dia")

Conclusion: checkpointing cada 100-500 steps no es opcional — es survival.

import os
import torch
import torch.distributed as dist
from torch.distributed.checkpoint import FileSystemCheckpoint
from torch.distributed.checkpoint import save, load

class DistributedCheckpointManager:
    def __init__(self, model, optimizer, save_dir, save_interval=100):
        self.model = model
        self.optimizer = optimizer
        self.save_dir = save_dir
        self.save_interval = save_interval
        self.step = 0
        
        # Filesystem checkpoint — guardara en el filesystem distribuidos
        self.checkpoint_io = FileSystemCheckpoint()
    
    def save_checkpoint(self, step, metadata=None):
        """
        Guardar estado distribuido.
        Solo el rank 0 coordina, pero todos los ranks guardan su parte.
        """
        state = {
            "model": self.model.state_dict(),
            "optimizer": self.optimizer.state_dict(),
            "step": step,
            "metadata": metadata or {}
        }
        
        # Guardar simultaneamente desde todos los ranks
        save(
            state,
            checkpoint_id=self.save_dir,
            checkpoint_io=self.checkpoint_io,
        )
        
        if dist.get_rank() == 0:
            print(f"Checkpoint saved at step {step}")
    
    def load_checkpoint(self, path):
        """
        Cargar checkpoint — puede ser desde cualquier paso guardado.
        """
        state = {
            "model": self.model.state_dict(),
            "optimizer": self.optimizer.state_dict(),
        }
        
        load(state, checkpoint_id=path, checkpoint_io=self.checkpoint_io)
        
        self.model.load_state_dict(state["model"])
        self.optimizer.load_state_dict(state["optimizer"])
        self.step = state["step"]
        
        return self.step

# Usage en training loop

Elasticity — Escalar Segun Necesidad

A escala, no todos los pasos de entrenamiento requieren la misma capacidad. Hay momentos donde puedes escalar horizontalmente (mas GPUs) y otros donde no tiene sentido.

# Para servicios que deben escalar a millones de requests:
class AutoScalingService:
    """
    Patron de auto-scaling para inference.
    Monitorea latency y hace scale up/down.
    """
    def __init__(self, model, min_replicas=1, max_replicas=100):
        self.model = model
        self.min_replicas = min_replicas
        self.max_replicas = max_replicas
        
        self.current_replicas = min_replicas
        self.pending_replicas = 0
        
        # Metrics
        self.latency_p50 = 0
        self.latency_p99 = 0
        self.queue_length = 0
        self.error_rate = 0
    
    def should_scale(self):
        """
        Decision de scale basada en metricas de latency y queue.
        """
        target_latency_p99 = 1000  # ms
        
        # Scale up si latency esta alto o cola esta creciendo
        if (self.latency_p99 > target_latency_p99 or 
            self.queue_length > self.current_replicas * 10):
            
            new_replicas = min(
                self.current_replicas * 2,
                self.max_replicas
            )
            return "scale_up", new_replicas
        
        # Scale down si latency es bajo y hay capacidad extra
        if (self.latency_p99 < target_latency_p99 * 0.5 and
            self.queue_length < self.current_replicas * 2 and
            self.current_replicas > self.min_replicas):
            
            return "scale_down", self.current_replicas // 2
        
        return "maintain", self.current_replicas
    
    def scale(self, action, target_replicas):
        if action == "scale_up":
            self.pending_replicas = target_replicas
            # Lanzar nuevas replicas
            self._launch_replicas(target_replicas - self.current_replicas)
            self.current_replicas = target_replicas
        
        elif action == "scale_down":
            # Terminar replicas innecesarias
            self._terminate_replicas(self.current_replicas - target_replicas)
            self.current_replicas = target_replicas

Metrics de Scale

Cuando entrenas a escala, necesitas saber exactamente que esta pasando. Métricas clave:

# Framework para metrics de entrenamiento distribuido
class TrainingMetrics:
    """
    Metrics criticos para entrenamiento a escala.
    """
    
    def __init__(self):
        self.metrics = {
            # Throughput
            "samples_per_second": 0,
            "tokens_per_second": 0,
            "steps_per_second": 0,
            
            # Eficiencia de GPU
            "gpu_utilization": [],
            "gpu_memory_used": [],
            "gpu_memory_total": [],
            
            # Comunicación
            "comm_overhead_ms": 0,
            "sync_time_ms": 0,
            
            # Loss y entrenamiento
            "loss": [],
            "learning_rate": [],
            "gradient_norm": [],
            
            # Salud del sistema
            "node_failures": 0,
            "oom_events": 0,
            "network_issues": 0,
        }
    
    def compute_throughput(self, world_size, step_time, batch_size):
        """
        Calcular throughput efectivo.
        """
        self.metrics["samples_per_second"] = (world_size * batch_size) / step_time
        self.metrics["steps_per_second"] = 1 / step_time
        
        # Tokens por segundo (asumiendo sequence_length fijo)
        seq_length = 2048
        tokens_per_step = world_size * batch_size * seq_length
        self.metrics["tokens_per_second"] = tokens_per_step / step_time
    
    def compute_communication_efficiency(self, ddp):
        """
        Que tan bien esta重叠 computation y communication.
        """
        # DDP metrics automaticos
        for name, metric in ddp.metrics.items():
            if "comm" in name:
                self.metrics["comm_overhead_ms"] = metric.avg
    
    def log(self, step, world_size):
        """
        Loggear metrics para monitoreo (Weights & Biases, etc.)
        """
        import wandb
        
        wandb.log({
            # Throughput
            "throughput/samples_per_second": self.metrics["samples_per_second"],
            "throughput/tokens_per_second": self.metrics["tokens_per_second"],
            
            # Eficiencia
            "efficiency/gpu_memory_pct": sum(self.metrics["gpu_memory_used"]) / 
                                         sum(self.metrics["gpu_memory_total"]),
            "efficiency/comm_overhead_ms": self.metrics["comm_overhead_ms"],
            
            # Training
            "train/loss": self.metrics["loss"][-1],
            "train/gradient_norm": self.metrics["gradient_norm"][-1],
            
            # Salud
            "health/node_failures_total": self.metrics["node_failures"],
            "health/oom_events_total": self.metrics["oom_events"],
        }, step=step)

Errores Comunes

Error 1: No Disenar Para Fallo Desde el Inicio

El problema: A escala, los fallos no son excepciones — son la norma. Si diseñas tu sistema asumiendo que todas las GPUs van a funcionar perfectamente durante semanas, vas a perder dias o semanas de entrenamiento cuando fallen.

La solucion: Failure is first-class citizen. Cada componente debe manejar fallos.

Error 2: Ignorar el Costo de Scale

El problema: Linear scaling no significa linear cost. Doblar GPUs no significa doblar el costo — hay overhead de comunicación, coordinación, y fall probability.

La solucion: Calcula el costo real antes de escalar. Utilidad marginal decreciente.

# Costo real de entrenamiento distribuido
def calculate_real_cost():
    """
    Calculo de costo que la mayoria ignora.
    """
    
    # Costo base: 8x A100 80GB en cloud
    cost_per_gpu_hour = 3.69  # AWS p4d.24xlarge on-demand
    
    # Setup
    n_gpus = 8
    training_hours = 24 * 90  # 90 dias
    
    base_cost = n_gpus * cost_per_gpu_hour * training_hours
    
    print(f"Costo base: ${base_cost:,.0f}")
    
    # Overheads que la gente no considera:
    
    # 1. GPU-to-GPU communication overhead
    # Con 8 GPUs vs 16 GPUs en mismo nodo:
    # - Mismo nodo: NVLink, ~900 GB/s bidireccional
    # - Multi-nodo: necesita red, ~100 GB/s maximo
    communication_overhead_pct = 15  # Estimado para multi-nodo
    
    effective_cost_multiplier = 1 / (1 - communication_overhead_pct/100)
    adjusted_cost = base_cost * effective_cost_multiplier
    
    print(f"Costo con overhead komunikacji: ${adjusted_cost:,.0f}")
    
    # 2. Fallos y re-entrenamiento
    # Con 1000 GPUs y 90 dias:
    expected_failures = calculate_expected_failures(
        n_gpus=1000,
        uptime_per_gpu=0.999,
        training_days=90,
        steps_per_day=1000
    )
    
    # Cada fallo = reentrenar desde checkpoint
    # Si pierdes 500 steps cada vez y tienes 100 fallos:
    lost_compute_hours = expected_failures * 500 * n_gpus / steps_per_day / 3600
    
    # Costo de fallo: $X por hora de GPU idle durante recovery
    failure_cost = lost_compute_hours * cost_per_gpu_hour * n_gpus
    
    print(f"Costo estimado por fallos: ${failure_cost:,.0f}")
    print(f"Total real: ${base_cost + adjusted_cost - base_cost + failure_cost:,.0f}")
    
    # 3. Opportunity cost: si fallas 3 semanas, perdiste 3 semanas de time-to-market
    time_to_market_value = 1000000  # $1M por semana de delay
    
    print(f"Costo de delay: ${3 * time_to_market_value:,.0f}")

def calculate_expected_failures(n_gpus, uptime_per_gpu, training_days, steps_per_day):
    """
    Calcular numero esperado de fallos durante entrenamiento largo.
    """
    import math
    
    # Probabilidad de que una GPU sobreviva todo el entrenamiento
    steps_total = training_days * steps_per_day
    p_survive = uptime_per_gpu ** steps_total
    
    # Probabilidad de que al menos una falle
    p_any_failure = 1 - p_survive ** n_gpus
    
    # Numero esperado de fallos (asumiendo distribucion geometrica)
    # El tiempo entre fallos sigue distribucion exponencial
    mtbf_hours = 1 / (1 - uptime_per_gpu) / steps_per_day * 24
    
    expected_failures = training_days * 24 / mtbf_hours * n_gpus
    
    return expected_failures

        for batch_size in batch_sizes:
            # Crear modelo dummy del tamano correcto
            model = create_dummy_model(n_params=model_size_m * 1e6)
            model = model.cuda()
            model = DDP(model, device_ids=[rank])
            
            # Inputs dummy
            inputs = torch.randn(batch_size, 512).cuda()
            
            # Warmup
            for _ in range(10):
                outputs = model(inputs)
                loss = outputs.sum()
                loss.backward()
            
            # Benchmark
            torch.cuda.synchronize()
            
            start = torch.cuda.Event(enable_timing=True)
            end = torch.cuda.Event(enable_timing=True)
            
            start.record()
            
            for _ in range(100):
                outputs = model(inputs)
                loss = outputs.sum()
                loss.backward()
            
            end.record()
            torch.cuda.synchronize()
            
            elapsed_ms = start.elapsed_time(end)
            
            if rank == 0:
                results.append({
                    "model_size_m": model_size_m,
                    "batch_size": batch_size,
                    "world_size": world_size,
                    "throughput_samples_per_sec": batch_size * 100 / (elapsed_ms/1000),
                    "elapsed_ms": elapsed_ms,
                })
    
    return results

Resumen

Hoy cubrimos ML Infrastructure at Scale — el mundo que existe cuando pasas de experimentacion a produccion real:

  • DDP te permite entrenar en multiples GPUs sincronizando gradientes automaticamente
  • FSDP divide el modelo entre GPUs cuando no cabe en una sola
  • Checkpointing no es opcional — fallos son inevitables, tienes que estar preparado
  • Elasticity te permite escalar arriba y abajo segun necesidad
  • Metrics te dan visibilidad sobre throughput, eficiencia y salud del sistema

Los tres errores mas comunes:

  1. No disenar para fallo — a escala, todo falla siempre, hay que asumirlo y manejarlo
  2. Ignorar el costo real de scale — no es linear, hay overheads que multiplican el costo
  3. No hacer benchmarks — lo que funciona en 1 GPU puede no funcionar en 1000

Eso es todo por hoy, parcero. En Colombia decimos "el que no conoce el monte, que se prepare el lomo." En ML, si no conoces como escala tu infrastructure, te va a tocar cargar con el peso de los fallos en produccion. Nos vemos manana!
