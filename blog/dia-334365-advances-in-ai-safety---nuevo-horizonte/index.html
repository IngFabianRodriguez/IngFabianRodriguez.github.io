🔬 Por Qué LLM Observability Es Crítico

En sistemas tradicionales de software, tienes logs, metrics, y traces como los tres pilares de observabilidad. Con LLMs, introduces una nueva dimensión: el modelo es un黑盒 con estado latente. No puedes simplemente agregar un logger y esperar entender qué pasó.

El Problema Fundamental

Una sola request a un LLM-powered system puede involucrar:

  • Múltiples embeddings (vector search)
  • Múltiples llamadas a APIs externas (tools)
  • Múltiples turns de conversación con context
  • Retrieval de múltiples chunks
  • Parsing y post-processing de la respuesta

Cuando algo sale mal — latencia alta, respuestas质量的下降, o crashes — sin tracing distribuidos, estás ciega.

La Math de Distributed Tracing

Un trace representa el flujo completo de una request a través de múltiples servicios. Cada span mide una unidad de trabajo:

trace_id = SHA256(operation_id + timestamp + inputs_hash)

La latencia total es la suma de los spans seriales más el minimum spanning tree de los paralelos:

latency_total = max(span_serial) + max(tree(span_parallel))

Con OpenTelemetry, cada span tiene:

  • trace_id: Identificador único del trace completo
  • span_id: Identificador único de este span
  • parent_id: Span padre (None si root)
  • start_time, end_time: Timestamps
  • attributes: Key-value pairs metadata
  • events: Timestamped annotations

Fix — OpenTelemetry Tracing desde Cero:

# ✅ BIEN: Tracing completo con OpenTelemetry
from opentelemetry import trace
from opentelemetry.sdk.trace import TracerProvider
from opentelemetry.sdk.trace.export import BatchSpanProcessor, ConsoleSpanExporter
from opentelemetry.sdk.resources import Resource
from opentelemetry.semconv.resource import ResourceAttributes
from opentelemetry.trace import Status, StatusCode

# 1. Setup del provider (once at app startup)
resource = Resource.create({
    ResourceAttributes.SERVICE_NAME: "rag-service",
    ResourceAttributes.SERVICE_VERSION: "1.0.0",
})

provider = TracerProvider(resource=resource)
processor = BatchSpanProcessor(ConsoleSpanExporter())  # Usa OTLP en prod
provider.add_span_processor(processor)
trace.set_tracer_provider(provider)

tracer = trace.get_tracer(__name__)

# 2. Instrument each operation as spans
def rag_query(question: str):
    with tracer.start_as_current_span("rag_query") as root_span:
        root_span.set_attribute("question", question)
        root_span.set_attribute("question.length", len(question))

        # Span: Embedding
        with tracer.start_as_current_span("embed_question") as embed_span:
            embed_start = time.time()
            query_embedding = embed_model.encode(question)
            embed_span.set_attribute("embedding.dimensions", len(query_embedding))
            embed_span.set_attribute("embedding.ms", (time.time() - embed_start) * 1000)

        # Span: Retrieval
        with tracer.start_as_current_span("retrieve_docs") as ret_span:
            ret_start = time.time()
            docs = vector_store.similarity_search(question, k=5)
            ret_span.set_attribute("retrieved.count", len(docs))
            ret_span.set_attribute("retrieval.ms", (time.time() - ret_start) * 1000)

        # Span: Context Building
        with tracer.start_as_current_span("build_context") as ctx_span:
            context = "\n".join([doc.page_content for doc in docs])
            ctx_span.set_attribute("context.length", len(context))

        # Span: LLM Call
        with tracer.start_as_current_span("llm_call") as llm_span:
            llm_start = time.time()
            response = llm.invoke(context + "\n\nQuestion: " + question)
            llm_span.set_attribute("llm.model", "gpt-4-turbo")
            llm_span.set_attribute("llm.ms", (time.time() - llm_start) * 1000)
            llm_span.set_attribute("llm.input_tokens", response.usage_metadata["input_tokens"])
            llm_span.set_attribute("llm.output_tokens", response.usage_metadata["output_tokens"])

        root_span.set_status(Status(StatusCode.OK))
        return response

Error 2: No Correlacionar Logs con Traces

Tener traces pero no correlacionarlos con logs es como tener上下文中，但没有连接。Los logs sin trace context son difíciles de filtrar:

# ❌ MAL: Logs sin correlación
def some_deep_function(data):
    logger.info("Processing data")  # ¿Qué request? ¿Qué trace?
    # ...
    logger.error("Failed")  # ¿Cuál trace falló?

Fix — Inyectar trace context en logs:

# ✅ BIEN: Logs con trace context correlacionado
from opentelemetry import trace

# Setup logging con trace context
class TraceContextFilter(logging.Filter):
    def filter(self, record):
        span = trace.get_current_span()
        if span and span.get_span_context().is_valid:
            ctx = span.get_span_context()
            record.trace_id = format(ctx.trace_id, '032x')
            record.span_id = format(ctx.span_id, '016x')
        else:
            record.trace_id = "no-trace"
            record.span_id = "no-span"
        return True

# En tu logging config
handler = logging.StreamHandler()
handler.addFilter(TraceContextFilter())
formatter = logging.Formatter(
    '%(asctime)s | %(levelname)s | trace_id=%(trace_id)s span_id=%(span_id)s | %(message)s'
)
handler.setFormatter(formatter)
logger.addHandler(handler)
logger.setLevel(logging.INFO)

# Ahora cada log tiene trace y span IDs
def some_deep_function(data):
    # Automatically includes trace_id and span_id
    logger.info("Processing data")  # trace_id=abc123 span_id=def456 | Processing data
    try:
        result = transform(data)
        logger.info(f"Transformed {len(result)} items")
    except Exception as e:
        logger.error(f"Failed to transform: {e}")  # Mismo trace_id para correlar

import pandas as pd

# 1. Recolecta spans y análisis
def analyze_traces(spans: list[ReadableSpan]):
    """Analiza traces para optimización"""

    df = pd.DataFrame([{
        'name': s.name,
        'duration_ms': (s.end_time - s.start_time) / 1e6,
        'trace_id': format(s.trace_id, '032x'),
        'attributes': dict(s.attributes),
    } for s in spans])

    # 2. Aggregated por operation type
    print("=== LATENCY BY OPERATION ===")
    print(df.groupby('name')['duration_ms'].agg(['mean', 'max', 'std', 'count']))

    # 3. P50, P95, P99 latencies
    print("\n=== PERCENTILES (ms) ===")
    for op in df['name'].unique():
        op_durations = df[df['name'] == op]['duration_ms'].sort_values()
        p50 = op_durations.iloc[int(len(op_durations) * 0.50)]
        p95 = op_durations.iloc[int(len(op_durations) * 0.95)]
        p99 = op_durations.iloc[int(len(op_durations) * 0.99)]
        print(f"{op}: P50={p50:.1f}, P95={p95:.1f}, P99={p99:.1f}")

    # 4. Correlación: retrieval distance vs accuracy
    if 'retrieved.count' in df['attributes'].iloc[0]:
        retrieval_counts = [a.get('retrieved.count', 0) for a in df['attributes']]
        print(f"\n=== RETRIEVAL STATS ===")
        print(f"Avg retrieved: {sum(retrieval_counts)/len(retrieval_counts):.1f}")

    # 5. Anomaly detection: spans > 2 std deviations
    print("\n=== ANOMALIES (>2 std) ===")

🔗 Tracing Cross-Service Calls

En arquitecturas microservicio, propagate el trace context entre servicios:

# ✅ TRACE PROPAGATION ENTRE SERVICIOS
from opentelemetry import trace
from opentelemetry.trace.propagation.tracecontext import TraceContextTextMapPropagator
from opentelemetry.context import Context

propagator = TraceContextTextMapPropagator()

# Servicio A: Extrae context Y lo inyecta en headers
def call_downstream_service(endpoint: str, payload: dict):
    # Get current span context
    ctx = trace.get_current_span().get_span_context()

    # Crear carrier con trace context
    carrier = {}
    propagator.inject(carrier, ctx=ctx)

    # Llamar downstream con headers
    headers = {
        "traceparent": carrier["traceparent"],  # W3C trace context
        "tracestate": carrier.get("tracestate", ""),
    }

    response = requests.post(endpoint, json=payload, headers=headers)
    return response

# Servicio B: Extrae trace context Y continua el trace
def handle_downstream_request(request):
    # Extraer context de headers entrantes
    carrier = {
        "traceparent": request.headers["traceparent"],
        "tracestate": request.headers.get("tracestate", ""),
    }

    ctx = propagator.extract(carrier)
    
    # Crear child span bajo el trace del caller
    with tracer.start_as_current_span(
        "downstream_handler",
        context=ctx,  # Link to parent trace!
    ) as span:
        span.set_attribute("downstream.endpoint", request.path)
        # Process...
        return {"status": "ok"}
