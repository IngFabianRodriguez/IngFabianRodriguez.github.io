class CircuitBreaker:
    """
    Circuit Breaker Pattern — previene cascading failures.
    
    Cuando un servicio downstream falla repetidamente, el circuit breaker
    se "abre" y corta el tráfico hacia ese servicio, dándole tiempo de
    recuperarse. Esto evita que una falla en el modelo se propagated a
    todo el sistema.
    
    Estados:
    - CLOSED: Todo funciona, el tráfico fluye normalmente
    - OPEN: Hay fallas, el tráfico se corta inmediatamente  
    - HALF_OPEN: Se prueba si el servicio se recuperó
    """
    
    def __init__(
        self,
        failure_threshold: int = 5,      # Veces que tiene que fallar para abrir
        recovery_timeout: int = 60,      # Segundos antes de probar otra vez
        expected_exception: type = Exception,
        name: str = "llm_circuit_breaker"
    ):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.expected_exception = expected_exception
        self.name = name
        
        self.failure_count = 0
        self.last_failure_time: Optional[datetime] = None
        self.state = "CLOSED"  # CLOSED, OPEN, HALF_OPEN
        
    def call(self, func: Callable, *args, **kwargs):
        """Ejecuta la función con circuit breaker."""
        
        if self.state == "OPEN":
            # Verificar si es tiempo de probar otra vez
            if self._should_attempt_reset():
                self.state = "HALF_OPEN"
                logger.info(f"{self.name}: Circuit breaker HALF_OPEN")
            else:
                raise CircuitBreakerOpenError(
                    f"{self.name}: Circuit breaker is OPEN. Call rejected."
                )
        
        try:
            result = func(*args, **kwargs)
            self._on_success()
            return result
        except self.expected_exception as e:
            self._on_failure()
            raise
    
    def _should_attempt_reset(self) -> bool:
        if self.last_failure_time is None:
            return True
        elapsed = (datetime.now() - self.last_failure_time).total_seconds()
        return elapsed >= self.recovery_timeout
    
    def _on_success(self):
        """Reset en caso de éxito."""
        if self.state == "HALF_OPEN":
            logger.info(f"{self.name}: Circuit breaker CLOSED after recovery")
        self.failure_count = 0
        self.state = "CLOSED"
    
    def _on_failure(self):
        """Incrementa contador de fallas."""
        self.failure_count += 1
        self.last_failure_time = datetime.now()
        
        if self.failure_count >= self.failure_threshold:
            self.state = "OPEN"
            logger.warning(
                f"{self.name}: Circuit breaker OPEN after {self.failure_count} failures"
            )


class CircuitBreakerOpenError(Exception):
    """Excepción cuando el circuit breaker está abierto."""
    pass

📊 Monitoreo: Lo Que No Mides, No Puedes Mejorar

Aquí está la verdad incómoda: el 80% de los equipos de ML que deployan LLMs no tienen monitoreo adecuados. No saben cuántos requests están sirviendo, cuál es su latencia real, ni cuándo el modelo está empezando a degradarse. Esto es como manejar sin tablero de instrumentos — eventualmente vas a chocar.

"""
LLM Production Monitoring System
================================
Monitoreo completo para sistemas LLM en producción.
"""

from dataclasses import dataclass, field
from typing import Dict, List, Optional, Tuple
from collections import deque
from datetime import datetime, timedelta
import statistics
import threading


@dataclass
class LatencyWindow:
    """Ventana temporal para tracking de latencia."""
    window_size_seconds: int = 60
    latencies: deque = field(default_factory=deque)
    timestamps: deque = field(default_factory=deque)
    
    def add(self, latency_ms: float):
        """Agrega una medición de latencia."""
        now = datetime.now()
        self.latencies.append(latency_ms)
        self.timestamps.append(now)
        self._clean_old()
    
    def _clean_old(self):
        """Remueve datos fuera de la ventana."""
        cutoff = datetime.now() - timedelta(seconds=self.window_size_seconds)
        while self.timestamps and self.timestamps[0] < cutoff:
            self.latencies.popleft()
            self.timestamps.popleft()
    
    def percentiles(self) -> Tuple[float, float, float]:
        """Calcula percentiles p50, p95, p99."""
        if not self.latencies:
            return 0.0, 0.0, 0.0
        sorted_latencies = sorted(self.latencies)
        n = len(sorted_latencies)
        return (
            sorted_latencies[int(n * 0.50)],
            sorted_latencies[int(n * 0.95)],
            sorted_latencies[int(n * 0.99)]
        )


class LLMPrometheusMetrics:
    """
    Sistema de métricas estilo Prometheus para LLMs.
    
    Recolecta y calcula:
    - Latencia (p50, p95, p99)
    - Throughput (requests/segundo)
    - Error rate
    - Token usage
    - Cost por request
    """
    
    # Ventana de 5 minutos para métricas
    WINDOW_SECONDS = 300
    
    def __init__(self):
        self.request_count = 0
        self.error_count = 0
        self.total_tokens = 0
        self.cost_total = 0.0
        
        self.latency_window = LatencyWindow(window_size_seconds=self.WINDOW_SECONDS)
        self._lock = threading.Lock()
        
        # Model pricing (USD por 1M tokens)
        self.model_pricing = {
            "gpt-4o": {"input": 5.0, "output": 15.0},
            "gpt-4o-mini": {"input": 0.15, "output": 0.60},
            "claude-3-5-sonnet": {"input": 3.0, "output": 15.0},
            "claude-3-5-haiku": {"input": 0.80, "output": 4.0},
        }
    
    def record_request(
        self,
        latency_ms: float,
        tokens_used: int,
        model: str,
        success: bool = True,
        error_type: Optional[str] = None
    ):
        """Registra una request completada."""
        with self._lock:
            self.request_count += 1
            self.total_tokens += tokens_used
            self.latency_window.add(latency_ms)
            
            # Calcular costo
            input_tokens = int(tokens_used * 0.3)  # Rough estimate
            output_tokens = int(tokens_used * 0.7)
            
            if model in self.model_pricing:
                pricing = self.model_pricing[model]
                cost = (input_tokens / 1_000_000) * pricing["input"]
                cost += (output_tokens / 1_000_000) * pricing["output"]
                self.cost_total += cost
            
            if not success:
                self.error_count += 1
    
    def get_metrics(self) -> Dict:
        """Retorna todas las métricas actuales."""
        with self._lock:
            p50, p95, p99 = self.latency_window.percentiles()
            
            error_rate = 0.0
            if self.request_count > 0:
                error_rate = self.error_count / self.request_count
            
            # Requests por segundo en la ventana
            requests_in_window = len(self.latency_window.latencies)
            throughput = requests_in_window / self.WINDOW_SECONDS
            
            return {
                "request_count": self.request_count,
                "error_count": self.error_count,
                "error_rate": error_rate,
                "total_tokens": self.total_tokens,
                "cost_total_usd": self.cost_total,
                "latency_p50_ms": p50,
                "latency_p95_ms": p95,
                "latency_p99_ms": p99,
                "throughput_rps": throughput,
                "cost_per_request_usd": (
                    self.cost_total / self.request_count 
                    if self.request_count > 0 else 0
                )
            }
    
    def export_prometheus_format(self) -> str:
        """Exporta métricas en formato Prometheus."""
        metrics = self.get_metrics()
        
        lines = [
            "# HELP llm_requests_total Total number of LLM requests",
            "# TYPE llm_requests_total counter",
            f'llm_requests_total {metrics["request_count"]}',
            "",
            "# HELP llm_errors_total Total number of failed requests",
            "# TYPE llm_errors_total counter", 
            f'llm_errors_total {metrics["error_count"]}',
            "",
            "# HELP llm_error_rate Current error rate",
            "# TYPE llm_error_rate gauge",
            f'llm_error_rate {metrics["error_rate"]:.4f}',
            "",
            "# HELP llm_latency_p50 Latency p50 in milliseconds",
            "# TYPE llm_latency_p50 gauge",
            f'llm_latency_p50 {metrics["latency_p50_ms"]:.2f}',
            "",
            "# HELP llm_latency_p95 Latency p95 in milliseconds",
            "# TYPE llm_latency_p95 gauge",
            f'llm_latency_p95 {metrics["latency_p95_ms"]:.2f}',
            "",
            "# HELP llm_latency_p99 Latency p99 in milliseconds",
            "# TYPE llm_latency_p99 gauge",
            f'llm_latency_p99 {metrics["latency_p99_ms"]:.2f}',
            "",
            "# HELP llm_throughput Requests per second",
            "# TYPE llm_throughput gauge",
            f'llm_throughput {metrics["throughput_rps"]:.2f}',
            "",
            "# HELP llm_cost_total Total cost in USD",
            "# TYPE llm_cost_total counter",
            f'llm_cost_total {metrics["cost_total_usd"]:.4f}',
        ]
        
        return "\n".join(lines)

🧮 La Matemática de la Confiabilidad

Vamos a hablar de algo que la mayoría de los equipos de ML ignoran: los números detrás de la confiabilidad. Porque en producción, los números importan.

MTBF y MTTR: Las Métricas que Definen tu Sistema

$$MTBF = \frac{\text{Tiempo total de operación}}{\text{Número de fallas}}$$

$$MTTR = \frac{\text{Tiempo total de reparación}}{\text{Número de reparaciones}}$$

Disponibilidad del Sistema

$$A = \frac{MTBF}{MTBF + MTTR}$$

Ejemplo práctico:

  • Si tu sistema tiene un MTBF de 720 horas (30 días) y un MTTR de 1 hora
  • $$A = \frac{720}{720 + 1} = 0.9986 = 99.86\%$$
  • Eso son aproximadamente 1.2 horas de downtime al mes

Relación Entre Número de Réplicas y Disponibilidad

$$A_{total} = 1 - (1 - A_{single})^n$$

Donde $n$ es el número de réplicas independientes:

  • 1 réplica: 99.0% → 87.6 horas de downtime/mes
  • 2 réplicas: 99.99% → 4.4 horas de downtime/mes
  • 3 réplicas: 99.999% → 26 minutos de downtime/mes

"En producción, la diferencia entre 99% y 99.9% de disponibilidad son 8 horas de downtime al mes. Esa diferencia tiene precio."

⚠️ Los 3 Errores Críticos en LLM Production Systems

Error #1: No Diseñar para Producción Desde elInicio

El problema: La mayoría de los equipos construyen su LLM system como si fuera un script de Jupyter. Pegan el API key, hacen unas llamadas, funcionan en pruebas, y deployan. Pero cuando el tráfico real llega, el sistema no aguanta.

Los problemas específicos incluyen:

  • Sin timeouts — si el modelo no responde, la request cuelga indefinidamente
  • Sin circuit breakers — cuando el modelo falla, todas las requests fallan
  • Sin rate limiting — un spike de tráfico tumba el servicio
  • Sin graceful degradation — cuando el LLM está down, todo está down

El fix:

"""
Production LLM Client con todos los patrones necesarios.
"""

import asyncio
from typing import Optional
import httpx
from circuit_breaker import CircuitBreaker


class ProductionLLMClient:
    """
    Cliente LLM production-ready con:
    - Timeouts configurables
    - Circuit breaker
    - Retry con exponential backoff
    - Rate limiting
    - Graceful degradation
    """
    
    def __init__(
        self,
        api_key: str,
        base_url: str,
        timeout_ms: int = 30000,
        max_retries: int = 3,
        circuit_breaker_threshold: int = 5,
        rate_limit_rpm: int = 60,
    ):
        self.api_key = api_key
        self.base_url = base_url
        self.timeout_ms = timeout_ms
        self.max_retries = max_retries
        
        # Circuit breaker para manejar fallas
        self.circuit_breaker = CircuitBreaker(
            failure_threshold=circuit_breaker_threshold,
            recovery_timeout=60,
            name="llm_client"
        )
        
        # Rate limiting
        self.rate_limit_rpm = rate_limit_rpm
        self.request_timestamps: deque = deque(maxlen=rate_limit_rpm)
        
        # HTTP client con timeouts
        self.client = httpx.AsyncClient(
            timeout=httpx.Timeout(timeout_ms / 1000),
            headers={"Authorization": f"Bearer {api_key}"}
        )
        
        # Fallback responses cuando el modelo falla
        self.fallback_responses = {
            "error": "Estamos experimentando alta demanda. Por favor intenta de nuevo en unos minutos.",
            "degraded": "El sistema está operando en modo degradado. Los resultados pueden tardar más de lo normal."
        }
    
    async def generate(
        self,
        prompt: str,
        model: str = "gpt-4o",
        use_fallback: bool = True
    ) -> Response:
        """
        Generate con todos los patterns de producción.
        """
        # 1. Rate limiting check
        if not self._check_rate_limit():
            return Response(
                request_id=generate_request_id(),
                content=self.fallback_responses["degraded"],
                model=model,
                tokens_used=0,
                latency_ms=0,
                timestamp=datetime.now(),
                success=False,
                error="rate_limit_exceeded"
            )
        
        # 2. Retry loop con exponential backoff
        last_error = None
        for attempt in range(self.max_retries):
            try:
                # Circuit breaker check
                response = await self.circuit_breaker.call(
                    self._call_llm_api,
                    prompt,
                    model
                )
                return response
                
            except CircuitBreakerOpenError:
                # Nointentar, return fallback inmediatamente
                if use_fallback:
                    return self._fallback_response(model, "circuit_breaker_open")
                raise
                
            except httpx.TimeoutException as e:
                last_error = e
                if attempt < self.max_retries - 1:
                    wait_time = 2 ** attempt  # Exponential backoff
                    await asyncio.sleep(wait_time)
                continue
                
            except httpx.HTTPStatusError as e:
                last_error = e
                if e.response.status_code >= 500:
                    # Error del servidor, retry tiene sentido
                    if attempt < self.max_retries - 1:
                        await asyncio.sleep(2 ** attempt)
                    continue
                else:
                    # Error del cliente (4xx), no retry
                    raise
        
        # Si llegamos aquí, todos los retries fallaron
        if use_fallback:
            return self._fallback_response(model, str(last_error))
        raise last_error
    
    async def _call_llm_api(self, prompt: str, model: str) -> Response:
        """Llama al API del LLM."""
        start = datetime.now()
        
        response = await self.client.post(
            f"{self.base_url}/chat/completions",
            json={
                "model": model,
                "messages": [{"role": "user", "content": prompt}],
                "temperature": 0.7,
                "max_tokens": 2048
            }
        )
        response.raise_for_status()
        
        data = response.json()
        latency = (datetime.now() - start).total_seconds() * 1000
        
        return Response(
            request_id=generate_request_id(),
            content=data["choices"][0]["message"]["content"],
            model=model,
            tokens_used=data["usage"]["total_tokens"],
            latency_ms=latency,
            timestamp=datetime.now(),
            success=True
        )
    
    def _fallback_response(self, model: str, error: str) -> Response:
        """Genera una respuesta de fallback cuando el sistema falla."""
        return Response(
            request_id=generate_request_id(),
            content=self.fallback_responses["error"],
            model=model,
            tokens_used=0,
            latency_ms=0,
            timestamp=datetime.now(),
            success=False,
            error=error
        )
    
    def _check_rate_limit(self) -> bool:
        """Verifica si estamos dentro del rate limit."""
        now = datetime.now()
        cutoff = now - timedelta(minutes=1)
        
        # Limpiar timestamps viejos
        while self.request_timestamps and self.request_timestamps[0] < cutoff:
            self.request_timestamps.popleft()
        
        return len(self.request_timestamps) < self.rate_limit_rpm

Error #3: No Manejar las Fallas Gritacios

El problema: Cuando algo falla, falla completo. No hay retry, no hay fallback, no hay nada. El usuario se queda viendo una pantalla de error o, peor, la request se pierde en el limbo.

El fix:

"""
Error Handler centralizado para LLM Production.
"""

from typing import Optional, Callable
from dataclasses import dataclass
from datetime import datetime
import logging

logger = logging.getLogger(__name__)


class LLMError(Exception):
    """Excepción base para errores de LLM."""
    pass


class ModelTimeoutError(LLMError):
    """El modelo tardó demasiado en responder."""
    pass


class ModelRateLimitError(LLMError):
    """El modelo está rate limited."""
    pass


class ModelContextLengthError(LLMError):
    """El prompt excede el context window del modelo."""
    pass


class ModelAPIError(LLMError):
    """Error retornado por el API del modelo."""
    pass


class ModelUnavailableError(LLMError):
    """El modelo no está disponible."""
    pass


@dataclass
class ErrorPolicy:
    """
    Política de manejo de errores.
    
    Define qué hacer cuando ocurre cada tipo de error.
    """
    max_retries: int
    retry_delay_seconds: float
    fallback_enabled: bool
    fallback_response: str
    alert_on_failure: bool


class ProductionErrorHandler:
    """
    Manejador centralizado de errores con políticas configurables.
    
    Cada tipo de error tiene una política específica que define:
    - Cuántas veces reintentar
    - Cuánto esperar entre reintentos
    - Si usar fallback
    - Si disparar alertas
    """
    
    ERROR_POLICIES = {
        ModelTimeoutError: ErrorPolicy(
            max_retries=3,
            retry_delay_seconds=2.0,
            fallback_enabled=True,
            fallback_response="La solicitud está tardando más de lo normal. Por favor intenta de nuevo.",
            alert_on_failure=True
        ),
        ModelRateLimitError: ErrorPolicy(
            max_retries=5,
            retry_delay_seconds=10.0,
            fallback_enabled=True,
            fallback_response="Estamos experimentando alta demanda. Por favor espera un momento.",
            alert_on_failure=True
        ),
        ModelContextLengthError: ErrorPolicy(
            max_retries=0,
            retry_delay_seconds=0,
            fallback_enabled=False,
            fallback_response="El mensaje es demasiado largo. Por favor reducelo.",
            alert_on_failure=False
        ),
        ModelAPIError: ErrorPolicy(
            max_retries=3,
            retry_delay_seconds=5.0,
            fallback_enabled=True,
            fallback_response="Hubo un error procesando tu solicitud. Por favor intenta de nuevo.",
            alert_on_failure=True
        ),
        ModelUnavailableError: ErrorPolicy(
            max_retries=10,
            retry_delay_seconds=30.0,
            fallback_enabled=True,
            fallback_response="El servicio está temporalmente unavailable. Estamos trabajando en resolverlo.",
            alert_on_failure=True
        ),
    }
    
    def __init__(self, alert_callback: Optional[Callable] = None):
        self.alert_callback = alert_callback
        self.error_log: list = []
    
    async def handle(
        self,
        error: Exception,
        operation: Callable,
        *args,
        **kwargs
    ):
        """
        Maneja un error usando la política apropiada.
        """
        error_type = type(error)
        policy = self.ERROR_POLICIES.get(
            error_type,
            ErrorPolicy(
                max_retries=1,
                retry_delay_seconds=1.0,
                fallback_enabled=True,
                fallback_response="Ocurrió un error inesperado. Por favor intenta de nuevo.",
                alert_on_failure=True
            )
        )
        
        # Log del error
        self._log_error(error_type, error, policy)
        
        # Alertar si es necesario
        if policy.alert_on_failure and self.alert_callback:
            self.alert_callback(error_type.__name__, str(error))
        
        # Reintentar si hay policy para ello
        last_error = error
        for attempt in range(policy.max_retries):
            if attempt > 0:
                await self._sleep(policy.retry_delay_seconds * (2 ** (attempt - 1)))
            
            try:
                return await operation(*args, **kwargs)
            except Exception as e:
                last_error = e
                continue
        
        # Si todos los reintentos fallaron, usar fallback
        if policy.fallback_enabled:
            logger.warning(
                f"All retries exhausted for {error_type.__name__}, "
                f"returning fallback response"
            )
            return policy.fallback_response
        
        # Si no hay fallback, lanzar la excepción
        raise last_error
    
    def _log_error(self, error_type: type, error: Exception, policy: ErrorPolicy):
        """Logea el error para debugging."""
        entry = {
            "timestamp": datetime.now().isoformat(),
            "error_type": error_type.__name__,
            "message": str(error),
            "max_retries": policy.max_retries,
            "fallback_enabled": policy.fallback_enabled
        }
        self.error_log.append(entry)
        logger.error(f"LLM Error: {entry}")
    
    async def _sleep(self, seconds: float):
        """Sleep asincrónico."""
        import asyncio
        await asyncio.sleep(seconds)

Y ahí lo tienen, parcero. Día 359 y hemos cubierto los fundamentos de construir LLM systems que no te van a dejar plantado en producción:

  • Arquitectura con Circuit Breakers: El sistema no se cae cuando una parte falla — corta el tráfico problemático y sigue funcionando.
  • Monitoreo Completo: Si no mides, no puedes mejorar. Latencia, error rate, costo — todo importa en producción.
  • Manejo de Errores: Los errores van a pasar. La diferencia entre un sistema que sobrevive y uno que se cae es cómo manejas esas fallas.

La matemática no miente: con 3 réplicas y circuit breakers, puedes pasar de 99% a 99.999% de disponibilidad. Eso es la diferencia entre dormir tranquilo o que te llamen a las 3 de la mañana.
