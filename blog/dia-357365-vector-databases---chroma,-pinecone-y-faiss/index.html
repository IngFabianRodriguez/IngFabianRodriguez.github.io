Por Que Vector Databases

Hagale pues, parcero. Dia 357 y continuamos con la serie de RAG e infrastructure de IA. Ayer hablamos de como hacer retrieval de documentos. Hoy nos metemos con las vector databases: las herramientas que hacen posible el almacenamiento y busqueda eficiente de embeddings a escala.

Si RAG es el motor, las vector databases son el tanque de gasolina. Sin ellas, el retrieval seria lentisimo e inmanejable.

Que es un Vector Database

Una vector database es una base de datos optimizada para almacenar embeddings (vectores numericos de alta dimensionalidad) y realizar busquedas por similitud de manera eficiente.

Cuando tienes 10 millones de embeddings de 1536 dimensiones cada uno, una busqueda lineal (calcular similitud con todos) toma segundos. Una vector database usa indices especiales para hacer esto en milisegundos.

FAISS: El Estandar de Facebook

import faiss
import numpy as np

# === CREAR INDICE FAISS ===
def create_faiss_index(embeddings: np.ndarray, metric: str = "l2") -> faiss.Index:
    """
    Crea un indice FAISS para busqueda por similitud.
    
    embeddings: matriz de shape (n, d) donde n=num docs, d=dimensiones
    metric: "l2" para distancia euclidiana, "ip" para producto interno (coseno sim)
    """
    n, d = embeddings.shape
    print(f"Creando indice para {n} vectores de dimension {d}")
    
    # Normalizar para similitud coseno si usamos producto interno
    if metric == "ip":
        faiss.normalize_L2(embeddings)
    
    # Index Flat: busqueda exacta pero lenta para grandes datasets
    # Bueno para menos de 1M vectores
    if n < 100_000:
        index = faiss.IndexFlatIP(d) if metric == "ip" else faiss.IndexFlatL2(d)
        index.add(embeddings)
        print(f"Index Flat creado con {index.ntotal} vectores")
        return index
    
    # Para datasets mas grandes: IVF (Inverted File Index)
    # Divide el espacio en nlist clusters
    nlist = min(4096, n // 100)  # numero de clusters
    quantizer = faiss.IndexFlatIP(d) if metric == "ip" else faiss.IndexFlatL2(d)
    index = faiss.IndexIVFFlat(quantizer, d, nlist)
    
    # Entrenar el indice (necesario para IVF)
    index.train(embeddings)
    index.add(embeddings)
    print(f"Index IVF creado con {index.ntotal} vectores en {nlist} clusters")
    
    return index


def search_faiss(index: faiss.Index, query: np.ndarray, k: int = 5) -> tuple:
    """
    Busca los k vecinos mas cercanos.
    
    Returns:
        distances: distancias a los k vecinos
        indices: indices de los k vecinos en el dataset original
    """
    distances, indices = index.search(query.reshape(1, -1), k)
    return distances[0], indices[0]

Chroma: La Opcion Local y Simple

# === CHROMA (Instalar: pip install chromadb) ===
import chromadb
from chromadb.config import Settings

# === CONFIGURACION ===
def setup_chroma(persist_directory: str = "./chroma_db"):
    """Configura Chroma con persistencia local"""
    client = chromadb.Client(Settings(
        persist_directory=persist_directory,
        anonymized_telemetry=False
    ))
    return client


def create_collection(client: chromadb.Client, name: str = "documentos"):
    """Crea o obtiene una coleccion"""
    collection = client.get_or_create_collection(
        name=name,
        metadata={"description": "Coleccion de documentos para RAG"}
    )
    return collection


def add_documents_to_chroma(collection, documents: list, embeddings: list, ids: list = None):
    """Agrega documentos con sus embeddings"""
    if ids is None:
        ids = [f"doc_{i}" for i in range(len(documents))]
    
    collection.add(
        documents=documents,
        embeddings=embeddings,
        ids=ids
    )
    print(f"Agregados {len(documents)} documentos a Chroma")


def search_chroma(collection, query_embedding: list, n_results: int = 5):
    """Busca documentos similares"""
    results = collection.query(
        query_embeddings=[query_embedding],
        n_results=n_results
    )
    return results

Pinecone: Vector Database como Servicio

# === PINECONE (Instalar: pip install pinecone-client) ===
import pinecone

# === CONFIGURACION ===
def setup_pinecone(api_key: str, environment: str = "us-east-1"):
    """Inicializa conexion a Pinecone"""
    pinecone.init(api_key=api_key, environment=environment)
    return pinecone


def create_index(name: str, dimension: int = 1536, metric: str = "cosine"):
    """
    Crea un indice en Pinecone.
    
    dimension: depende del modelo de embedding (OpenAI ada-002 = 1536)
    metric: "cosine", "euclidean", o "dotproduct"
    """
    if name in pinecone.list_indexes():
        print(f"Index {name} ya existe")
        return pinecone.Index(name)
    
    pinecone.create_index(
        name=name,
        dimension=dimension,
        metric=metric,
        pods=1,  # Numero de pods (escala con mas datos)
        replicas=1,
        shards=1
    )
    
    print(f"Index {name} creado")
    return pinecone.Index(name)


def upsert_vectors(index, vectors: dict, namespace: str = ""):
    """
    Inserta o actualiza vectores.
    
    vectors: dict con {id: {"values": embedding, "metadata": {...}}}
    namespace: namespace opcional para multitenant
    """
    index.upsert(vectors, namespace=namespace)
    print(f"Upserted {len(vectors)} vectors")


def query_pinecone(index, vector: list, top_k: int = 5, filter_dict: dict = None):
    """Busca vectores similares"""
    query_response = index.query(
        top_k=top_k,
        vector=vector,
        include_values=True,
        include_metadata=True,
        filter=filter_dict
    )
    return query_response

Comparacion y Cuando Usar Cada Uno

  • FAISS: Ideal para prototipado local y cuando ya tienes tu propia infrastructure. Es el mas rapido para datasets medianos pero requiere gestion manual.
  • Chroma: Perfecto para proyectos pequenos a medianos, prototipado rapido, y cuando quieres algo simple sin infraestructura cloud.
  • Pinecone: La opcion enterprise cuando necesitas escalar sin gestionar servidores, alta disponibilidad, y caracteristicas avanzadas como filtrado de metadatos.

Indices Avanzados en FAISS

# === INDICES MAS sofisticados ===

def create_product_quantization_index(embeddings: np.ndarray, m: int = 96):
    """
    Product Quantization (PQ) para compresion extrema.
    Reduce memoria de O(n*d) a O(n*m) donde m << d.
    
    m: numero de subvectores (debe ser divisible por d)
    """
    n, d = embeddings.shape
    
    # 1. Entrenar el quantizer
    # PQ divide cada vector en m subvectores de d/m dimensiones
    # y los cuantiza independientemente
    bytes_per_vector = m  # cada subvector se guarda en 1 byte
    pq = faiss.ProductQuantizer(d, m, bytes_per_vector)
    
    # Train
    pq.train(embeddings)
    
    # 2. Crear indice con el quantizer
    # IVF reduce el espacio de busqueda
    nlist = min(4096, n // 100)
    quantizer = faiss.IndexFlatIP(d)
    index = faiss.IndexIVFPQ(quantizer, d, nlist, m, 8)  # 8 bits por subvector
    
    index.train(embeddings)
    index.add(embeddings)
    
    return index, pq


def create_hnsw_index(d: int, m: int = 32, ef_construction: int = 200):
    """
    HNSW (Hierarchical Navigable Small World) para busqueda rapida y precisa.
    
    m: numero de conexiones por elemento
    ef_construction: tamano de la cola de busqueda durante construccion
    """
    index = faiss.IndexHNSWFlat(d, m)
    index.hnsw.efConstruction = ef_construction
    
    return index


def search_hnsw(index, query: np.ndarray, k: int = 5, ef_search: int = 128):
    """
    Busqueda en HNSW.
    
    ef_search: tamano de la cola de busqueda (mas = mejor precision, mas lento)
    """
    index.hnsw.efSearch = ef_search
    distances, labels = index.search(query.reshape(1, -1), k)
    return distances[0], labels[0]


# === BENCHMARK ===
def benchmark_indices(embeddings: np.ndarray, queries: np.ndarray, k: int = 10):
    """Compara velocidad y precision de diferentes indices"""
    n = len(embeddings)
    
    results = {}
    
    # Flat (baseline exacto)
    flat = faiss.IndexFlatIP(embeddings.shape[1])
    flat.add(embeddings)
    import time
    start = time.time()
    for q in queries:
        flat.search(q.reshape(1, -1), k)
    results["Flat (exact)"] = time.time() - start
    
    # IVF
    nlist = min(100, n // 1000)
    ivf = faiss.IndexIVFFlat(faiss.IndexFlatIP(embeddings.shape[1]), embeddings.shape[1], nlist)
    ivf.train(embeddings)
    ivf.add(embeddings)
    ivf.nprobe = 10
    start = time.time()
    for q in queries:
        ivf.search(q.reshape(1, -1), k)
    results["IVF (nprobe=10)"] = time.time() - start
    
    # HNSW
    hnsw = faiss.IndexHNSWFlat(embeddings.shape[1], 16)
    hnsw.add(embeddings)
    start = time.time()
    for q in queries:
        hnsw.search(q.reshape(1, -1), k)
    results["HNSW"] = time.time() - start
    
    print("Benchmark results:")
    for name, t in results.items():
        print(f"  {name}: {t:.3f}s")
    
    return results

Resumen y Cierre

Eso era Vector Databases, parcero. Las tres herramientas principales:

  • FAISS: Rapida, local, muchas opciones de indices. Ideal para prototipado y produccion cuando tienes infra.
  • Chroma: Simple, local, buena para proyectos pequenos y medianos. Muy popular para prototipar.
  • Pinecone: Servicio cloud gestionado, escala automaticamente, ideal para enterprise.

La eleccion depende de tu caso de uso especifico y escala. Para prototipar: Chroma. Para produccion mediana: FAISS. Para escala enterprise: Pinecone o servicios similares como Weaviate o Qdrant.
